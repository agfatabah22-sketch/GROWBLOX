local makeBackgroundOptimizer = (function()
-- Only the native scenic backdrop is hidden. Tiles, drops and game UI stay intact.
return function(player, app)
    local api={Status="Menunggu background world.",HiddenCount=0,EffectCount=0,HiddenPlantFlights=0}
    local closed,bound=false,false
    local pg,gui,world,backdrop,module,camera,tiles
    local links={}
    local originalVisible,native,nativeHidden
    local cachedModule,cachedClass
    local ticket,pending=0,false
    local changing=false
    local effects,flightRestores=nil,{}
    local refresh,queueNative
    local function active() return not closed and app.Alive~=false end
    local function isClass(object,class)
        if not object then return false end
        local ok,value=pcall(object.IsA,object,class)
        return ok and value==true
    end
    local function child(parent,name,class)
        if not parent then return end
        local ok,value=pcall(parent.FindFirstChild,parent,name)
        if ok and isClass(value,class) then return value end
    end
    local function clearEffects()
        if effects then
            for _,name in ipairs({"puff","place"}) do
                if rawget(effects.particles,name)==effects.overrides[name] then
                    rawset(effects.particles,name,effects.originals[name])
                end
            end
        end
        for frame,saved in pairs(flightRestores) do
            pcall(function()
                if frame.Parent==saved.parent then
                    frame.Visible=saved.visible
                    if saved.entry.t0==saved.appliedT0 then saved.entry.t0=saved.t0 end
                end
            end)
        end
        flightRestores={} effects=nil api.EffectCount=0
    end
    local function validEffects(binding)
        if type(binding)~="table" or not camera or not tiles or not gui then return false end
        local particles,class=binding.particles,binding.particleClass
        return type(particles)=="table" and type(class)=="table" and getmetatable(particles)==class
            and particles.parent==camera and type(particles.live)=="table" and type(particles.free)=="table"
            and type(class.puff)=="function" and type(class.place)=="function"
            and binding.plantParent==tiles and type(binding.plantFlights)=="table"
            and camera.Parent==world and tiles.Parent==camera
            and child(gui,"Particles","ModuleScript")~=nil
    end
    function api.BindFarmEffects(binding)
        if not active() then return false end
        if effects and validEffects(binding) and effects.particles==binding.particles
            and effects.particleClass==binding.particleClass and effects.plantFlights==binding.plantFlights
            and effects.plantParent==binding.plantParent then return true end
        clearEffects()
        if not validEffects(binding) then return false end
        local particles=binding.particles
        -- The observed native instance uses class methods. Do not replace an
        -- unrelated component's existing per-instance override.
        if rawget(particles,"puff")~=nil or rawget(particles,"place")~=nil then return false end
        effects={particles=particles,particleClass=binding.particleClass,plantFlights=binding.plantFlights,
            plantParent=binding.plantParent,originals={},overrides={}}
        for _,name in ipairs({"puff","place"}) do
            local noVisual=function() return nil end
            effects.overrides[name]=noVisual
            rawset(particles,name,noVisual)
        end
        api.EffectCount=2
        return true
    end
    function api.SuppressPlantFlights()
        if not active() or not effects or not validEffects(effects) then return 0 end
        for frame,saved in pairs(flightRestores) do
            local ok,parent=pcall(function() return frame.Parent end)
            if not ok or parent~=saved.parent then flightRestores[frame]=nil end
        end
        local hidden=0
        for _,entry in ipairs(effects.plantFlights) do
            local icon=type(entry)=="table" and entry.icon
            local frame=type(icon)=="table" and icon.frame
            if isClass(frame,"Frame") and frame.Parent==tiles and frame.ZIndex==14
                and type(entry.t0)=="number" and entry.t0==entry.t0 and math.abs(entry.t0)<math.huge
                and typeof(entry.from)=="Vector2" and typeof(entry.to)=="Vector2" and not flightRestores[frame] then
                local applied=os.clock()-1
                flightRestores[frame]={parent=tiles,visible=frame.Visible,entry=entry,t0=entry.t0,appliedT0=applied}
                frame.Visible=false entry.t0=applied
                hidden=hidden+1
            end
        end
        api.HiddenPlantFlights=api.HiddenPlantFlights+hidden
        return hidden
    end
    local function disconnect()
        for _,connection in ipairs(links) do pcall(connection.Disconnect,connection) end
        links={}
    end
    local function connect(signal,callback)
        if not signal then return end
        local ok,connection=pcall(signal.Connect,signal,callback)
        if ok then links[#links+1]=connection end
    end
    local function nativeMatches(value)
        return type(value)=="table" and value.parent==backdrop and value.world==world
            and type(value.setHidden)=="function"
    end
    local function restoreNative()
        if native and nativeMatches(native) and native.hidden==true then
            pcall(native.setHidden,native,nativeHidden==true)
        end
        native,nativeHidden=nil,nil
    end
    local function restoreBackdrop()
        changing=true
        restoreNative()
        if backdrop and originalVisible~=nil then
            pcall(function() backdrop.Visible=originalVisible end)
        end
        changing=false
        originalVisible=nil
    end
    local function hide()
        if not active() or not backdrop or changing then return end
        changing=true
        if native and nativeMatches(native) and native.hidden~=true then
            pcall(native.setHidden,native,true)
        end
        local ok=pcall(function() if backdrop.Visible~=false then backdrop.Visible=false end end)
        changing=false
        api.HiddenCount=ok and 1 or 0
        api.Status=ok and (native and native.hidden==true and "Background ringan: visual dan animasi native disembunyikan."
            or "Background ringan: visual disembunyikan.") or "Background belum tersedia."
    end
    local function adoptCurrent(class)
        if not active() or type(class)~="table" then return false end
        local current=class.current
        if not nativeMatches(current) then return false end
        if current~=native then
            changing=true
            restoreNative()
            native,nativeHidden=current,current.hidden
            changing=false
        end
        hide()
        return true
    end
    queueNative=function()
        if not active() or not backdrop or not module or pending then return end
        pending=true ticket=ticket+1
        local scheduled=ticket
        local targetModule,targetBackdrop,targetWorld=module,backdrop,world
        task.defer(function()
            if not active() or scheduled~=ticket then return end
            local class=cachedModule==targetModule and cachedClass or nil
            if not class then
                local ok,value=pcall(require,targetModule)
                if ok and type(value)=="table" then class=value end
            end
            if not active() or scheduled~=ticket or module~=targetModule
                or backdrop~=targetBackdrop or world~=targetWorld then return end
            pending=false
            if class then cachedModule,cachedClass=targetModule,class end
            if adoptCurrent(class) then return end
            -- The native constructor adds Backdrop before assigning class.current.
            -- One deferred retry is enough to cover that order without a poll loop.
            task.defer(function()
                if active() and scheduled==ticket and module==targetModule
                    and backdrop==targetBackdrop and world==targetWorld then adoptCurrent(class) end
            end)
        end)
    end
    local function watchChildren(object,predicate)
        if not object then return end
        local function changed(value)
            if active() and predicate(value) then refresh() end
        end
        connect(object.ChildAdded,changed)
        connect(object.ChildRemoved,changed)
    end
    refresh=function()
        if not active() then return end
        local ok,newPG=pcall(player.FindFirstChildOfClass,player,"PlayerGui")
        if not ok then newPG=nil end
        local newGUI=child(newPG,"GameGui","ScreenGui")
        local newWorld=child(newGUI,"World","Frame")
        local newBackdrop=child(newWorld,"Backdrop","Frame")
        local newModule=child(newGUI,"Backdrop","ModuleScript")
        local newCamera=child(newWorld,"Camera","Frame")
        local newTiles=child(newCamera,"Tiles","Frame")
        if bound and pg==newPG and gui==newGUI and world==newWorld and backdrop==newBackdrop and module==newModule
            and camera==newCamera and tiles==newTiles then
            hide() queueNative() return
        end
        disconnect()
        ticket=ticket+1 pending=false
        restoreBackdrop()
        if gui~=newGUI or world~=newWorld or camera~=newCamera or tiles~=newTiles then clearEffects() end
        pg,gui,world,backdrop,module=newPG,newGUI,newWorld,newBackdrop,newModule
        camera,tiles=newCamera,newTiles
        bound=true
        if module~=cachedModule then cachedModule,cachedClass=nil,nil end
        watchChildren(player,function(value) return isClass(value,"PlayerGui") end)
        watchChildren(pg,function(value) return value.Name=="GameGui" end)
        watchChildren(gui,function(value) return value.Name=="World" or value.Name=="Backdrop" end)
        watchChildren(world,function(value) return value.Name=="Backdrop" or value.Name=="Camera" end)
        watchChildren(camera,function(value) return value.Name=="Tiles" end)
        if backdrop then
            originalVisible=backdrop.Visible
            connect(backdrop:GetPropertyChangedSignal("Visible"),function()
                if changing or not active() then return end
                adoptCurrent(cachedClass)
                hide() queueNative()
            end)
            hide() queueNative()
        else
            api.HiddenCount=0 api.Status="Menunggu background world."
        end
    end
    function api.Close()
        if closed then return end
        closed=true ticket=ticket+1 pending=false
        disconnect()
        clearEffects()
        restoreBackdrop()
        api.HiddenCount=0 api.Status="Optimasi background ditutup."
    end
    refresh()
    return api
end

end)()

local makeFarmEngine = (function()
-- Crop row planner. The adapter owns native movement, reach, item and permission checks.
return function(adapter, options)
    options = options or {}
    local world = options.world or "AAXYOU"
    local farmMode = options.mode == "break_only" and "break_only" or "farm"
    local state = {
        mode = farmMode,
        running = false, closed = false, phase = "stopped", status = "Ready. Stand at the farm position, then START.",
        planted = 0, placed = 0, punches = 0, harvested = 0, blockIndex = 1, blockCount = 5,
        seeds = 0, blocks = 0, growing = 0, mature = 0, seedCount = farmMode == "break_only" and 0 or 21, occupied = 0,
        batchPhase = "planting", allowPickupTravel = false, restocking = false,
    }
    local api = { State = state }
    local revision, stepping, nextAction = 0, false, 0
    local mode, cursor = "fill", 1
    local cleanupPending, initialCleanupNeeded, overheadDirty, sweepTouched = false, true, false, false
    local activeBlockCount, activeSeedCount
    local activeRowKey, rowDirection, rowCursor, plantingStarted = nil, nil, nil, false
    local pending, attempts = {}, {}
    local resumeHint
    if type(options.resume) == "table" then
        resumeHint = {}
        for _,field in ipairs({"version","world","mode","rowKey","seedCount","blockCount","phase",
            "rowDirection","rowCursor","plantingStarted","blockCursor"}) do
            resumeHint[field] = options.resume[field]
        end
    end
    local actionInterval, confirmationTimeout, retryLimit = 0.22, 2.5, 3
    local suspendRequested=false
    local function canSuspend()
        return next(pending)==nil and not cleanupPending and not overheadDirty
            and not adapter.SeedOverflowActive and not adapter.TravelBusy
            and not (state.batchPhase=="planting" and plantingStarted)
            and not (state.batchPhase=="harvesting" and state.occupied>0)
    end
    local function finite(value)
        return type(value) == "number" and value == value and value > -math.huge and value < math.huge
    end
    local function blockCount(view) return view.blockCount == nil and 5 or view.blockCount end
    local function seedCount(view) return view.seedCount == nil and 21 or view.seedCount end
    local function rowKey(view)
        if farmMode == "break_only" then
            local first,last = view.blockSlots[1],view.blockSlots[blockCount(view)]
            return tostring(view.world)..":blocks:"..blockCount(view)..":"..tostring(first.x)..":"..tostring(first.y)
                ..":"..tostring(last.x)..":"..tostring(last.y)..":"..tostring(view.seedId)..":"..tostring(view.blockId)
        end
        local first,last=view.seedSlots[1],view.seedSlots[seedCount(view)]
        return tostring(view.world)..":"..seedCount(view)..":"..tostring(first.x)..":"..tostring(first.y)
            ..":"..tostring(last.x)..":"..tostring(last.y)..":"..tostring(view.seedId)..":"..tostring(view.blockId)
    end
    local function release()
        if type(adapter.Release) == "function" then pcall(adapter.Release) end
    end
    local function status(phase, message)
        state.phase, state.status = phase, message
    end
    local function stop(message)
        state.running = false
        state.allowPickupTravel = false
        state.restocking = false
        revision = revision + 1
        release()
        status("stopped", message or "Stopped manually.")
    end
    local function read()
        local ok, view = pcall(adapter.Snapshot)
        if not ok or type(view) ~= "table" then return nil, "Waiting for farm data." end
        if view.invalidWorld == true then
            return nil, tostring(view.message or "World, session, anchor or farm permission changed. Set the farm position again."), true
        end
        if view.world ~= nil and view.world ~= "" and view.world ~= world then
            return nil, "World changed. Farm is restricted to " .. world .. ".", true
        end
        if not view.ready or view.world ~= world then return nil, tostring(view.message or "Waiting for the world to load.") end
        if view.mode ~= nil and view.mode ~= farmMode then return nil, "Farm mode changed. Select the mode again before START.", true end
        if view.canFarm == false then return nil, tostring(view.message or "This position is not available for farming."), true end
        if view.atAnchor ~= true and view.canTravel ~= true then return nil, tostring(view.message or "Return to the saved farm position to continue.") end
        if view.canFarm ~= true or not finite(view.blocks) or view.blocks < 0 or type(view.blockSlots) ~= "table"
            or farmMode ~= "break_only" and (not finite(view.seeds) or view.seeds < 0 or type(view.seedSlots) ~= "table") then
            return nil, "Waiting for valid inventory / farm data."
        end
        local overheadCount = blockCount(view)
        if overheadCount ~= 3 and overheadCount ~= 5 then return nil, "Unsupported overhead block count; use three or five.", true end
        local rowCount = farmMode == "break_only" and 0 or seedCount(view)
        if farmMode ~= "break_only" then
        if not finite(rowCount) or rowCount % 1 ~= 0 or rowCount < 1 or rowCount > 99 then
            return nil, "Plant count must be a whole number from 1 to 99.", true
        end
        for i = 1, rowCount do
            local seed = view.seedSlots[i]
            if type(seed) ~= "table" then return nil, "Waiting for all " .. rowCount .. " seed slots." end
            if seed.state == "blocked" then
                return nil, "Slot " .. i .. " contains another object. Clear the farm area manually, then START.", true
            end
            if seed.state ~= "empty" and seed.state ~= "growing" and seed.state ~= "mature" then return nil, "Waiting for valid tree data." end
        end
        end
        for i = 1, overheadCount do
            local block = view.blockSlots[i]
            if type(block) ~= "table" then return nil, "Waiting for all overhead block slots." end
            if block.state == "blocked" then
                return nil, "Block slot " .. i .. " contains another object. Clear the farm area manually, then START.", true
            end
            if block.state ~= "empty" and block.state ~= "grass" then return nil, "Waiting for valid block data." end
        end
        return view
    end
    local function counts(view)
        state.blockCount, state.seedCount = blockCount(view), farmMode == "break_only" and 0 or seedCount(view)
        state.seeds, state.blocks, state.growing, state.mature, state.occupied = finite(view.seeds) and view.seeds or 0, view.blocks, 0, 0, 0
        state.seedName, state.blockName = view.seedName or "Seed", view.blockName or "Block"
        if farmMode == "break_only" then return end
        for i = 1, state.seedCount do
            if view.seedSlots[i].state == "growing" then state.growing = state.growing + 1 end
            if view.seedSlots[i].state == "mature" then state.mature = state.mature + 1 end
        end
        state.occupied = state.growing + state.mature
    end
    local function grassStockGate(view)
        return view.seedId==17 and view.blockId==16 and view.blocks>=30
    end
    local function readyToHarvest(view)
        return state.mature==seedCount(view) and not grassStockGate(view)
    end
    local function key(kind, i) return kind .. ":" .. i end
    local function settle(view, now)
        for id, action in pairs(pending) do
            local value = action.kind == "Place" and view.blockSlots[action.index].state or view.seedSlots[action.index].state
            local confirmed = action.kind == "Plant" and value ~= "empty"
                or action.kind == "Place" and value == "grass"
                or action.kind == "Harvest" and value ~= "mature"
            if confirmed then
                pending[id], attempts[id] = nil, nil
                if action.kind == "Plant" then state.planted = state.planted + 1
                elseif action.kind == "Place" then state.placed = state.placed + 1
                elseif action.kind == "Harvest" then state.harvested = state.harvested + 1 end
            elseif now >= action.expires then
                pending[id] = nil
                if (attempts[id] or 0) >= retryLimit then
                    stop(action.kind .. " slot " .. action.index .. " was not confirmed. Check the farm, then START.")
                    return false
                end
            end
        end
        return true
    end
    local function isPending(kind, i) return pending[key(kind, i)] ~= nil end
    local function prepare(kind, i, now, before)
        if type(adapter.Prepare) ~= "function" then return true end
        local ok, ready, message, fatal = pcall(adapter.Prepare, kind, i, now)
        if state.closed or not state.running or revision ~= before then return false end
        if not ok then stop("Farm travel paused: " .. tostring(ready):sub(1, 110)) return false end
        if fatal == true then stop(tostring(message or "The farm path is unavailable.")) return false end
        if ready ~= true then
            status("moving", tostring(message or "Moving to the farm target."))
            return false
        end
        return true
    end
    local function cleanup(now, before)
        if type(adapter.CleanupBlockDrops) == "function" then
            status("cleanup", "Collecting overhead block drops before the next batch.")
            local ok, ready, message, fatal = pcall(adapter.CleanupBlockDrops, now)
            if state.closed or not state.running or revision ~= before then return false end
            if not ok then stop("Block pickup paused: " .. tostring(ready):sub(1,110)) return false end
            if fatal == true then stop(tostring(message or "Block pickup permission changed.")) return false end
            if ready ~= true then
                cleanupPending = true
                status("cleanup", tostring(message or "Waiting for overhead block drops to be collected."))
                return false
            end
        end
        cleanupPending, initialCleanupNeeded, overheadDirty = false, false, false
        return true
    end
    local function finishSweep()
        mode = "fill"
        if sweepTouched then cleanupPending = true end
        sweepTouched = false
    end
    local function beginBatch(phase, view)
        state.batchPhase, plantingStarted = phase, false
        local n=seedCount(view)
        local left,right=view.seedSlots[1].distance,view.seedSlots[n].distance
        rowDirection = finite(right) and (not finite(left) or right<left) and -1 or 1
        rowCursor = rowDirection==1 and 1 or n
        if phase=="growing" then mode,cursor,state.blockIndex="fill",1,1 end
        if phase=="planting" or phase=="harvesting" then initialCleanupNeeded=true end
    end
    local function restoreProgress(view)
        local hint = resumeHint
        resumeHint = nil
        if not hint or hint.version ~= 1 or hint.world ~= world or hint.mode ~= farmMode
            or hint.rowKey ~= rowKey(view) or hint.blockCount ~= blockCount(view)
            or hint.seedCount ~= (farmMode == "break_only" and 0 or seedCount(view)) then return false end
        if farmMode == "break_only" then
            if hint.phase ~= "blocks" then return false end
            state.batchPhase = "blocks"
        else
            if hint.phase ~= "planting" and hint.phase ~= "growing" and hint.phase ~= "harvesting" then return false end
            if (hint.rowDirection ~= 1 and hint.rowDirection ~= -1) or not finite(hint.rowCursor)
                or hint.rowCursor % 1 ~= 0 or hint.rowCursor < 1 or hint.rowCursor > seedCount(view) then return false end
            state.batchPhase, rowDirection, rowCursor = hint.phase, hint.rowDirection, hint.rowCursor
            plantingStarted = hint.phase == "planting" and hint.plantingStarted == true
        end
        -- Inspect native blocks and clean leftovers anew; never restore a sent
        -- request, receipt timer, pickup movement or inventory counter.
        mode = "fill"
        cursor = finite(hint.blockCursor) and hint.blockCursor % 1 == 0
            and hint.blockCursor >= 1 and hint.blockCursor <= blockCount(view) and hint.blockCursor or 1
        state.blockIndex = cursor
        initialCleanupNeeded = true
        return true
    end
    local function send(kind, i, now, before)
        local fn = adapter[kind]
        if type(fn) ~= "function" then stop("Native " .. kind .. " action is unavailable.") return end
        local ok, sent, readyAt = pcall(fn, i)
        if state.closed or not state.running or revision ~= before then return end
        if ok and finite(readyAt) and readyAt > now then nextAction = readyAt
        else nextAction = now + actionInterval end
        if not ok then stop(kind .. " paused: " .. tostring(sent):sub(1, 110)) return end
        if sent ~= true then
            status("waiting", "Waiting for native " .. kind .. " readiness at slot " .. i .. ".")
            return false
        end
        if kind == "PunchBlock" then
            state.punches = state.punches + 1
            overheadDirty, sweepTouched = true, true
        else
            local id = key(kind, i)
            attempts[id] = (attempts[id] or 0) + 1
            pending[id] = { kind = kind, index = i, expires = now + confirmationTimeout }
        end
        return true
    end
    local function seedWork(view)
        local missing,plantingPending=0,false
        for i=1,seedCount(view) do
            if view.seedSlots[i].state=="empty" then
                if isPending("Plant",i) then plantingPending=true else missing=missing+1 end
            end
        end
        return missing,plantingPending
    end
    local function afterCleanup(now,before)
        -- Pickup can confirm a stack while cleanup finishes. Decide whether to
        -- begin harvesting from the resulting bag count, not the pre-pickup one.
        local view,waiting,blocked=read()
        if state.closed or not state.running or revision~=before then return end
        if not view then
            release()
            if blocked then stop(waiting) else status("waiting",waiting) end
            return
        end
        if blockCount(view)~=activeBlockCount or seedCount(view)~=activeSeedCount or rowKey(view)~=activeRowKey then
            stop("Farm layout changed during block cleanup. Press START at the current row.") return
        end
        counts(view)
        if not settle(view,now) then return end
        if finite(view.nextReadyAt) then nextAction=math.max(nextAction,view.nextReadyAt) end
        return view
    end
    local function seedOverflow(now,allowStart,before)
        if farmMode=="break_only" or type(adapter.ServiceSeedOverflow)~="function" then return false end
        local ok,busy,message,fatal=pcall(adapter.ServiceSeedOverflow,now,allowStart)
        if state.closed or not state.running or revision~=before then return true end
        if not ok then stop("Seed overflow paused: "..tostring(busy):sub(1,110)) return true end
        if fatal==true then stop(tostring(message or "Seed overflow permission changed.")) return true end
        if busy==true then
            -- Only the presentation phase changes. Resume hints keep the same
            -- crop batch, sweep direction and cursor during surplus recycling.
            status("seed_overflow",tostring(message or "Recycling excess Grass seeds; keeping 21."))
            return true
        end
        return false
    end
    local function step(now)
        state.allowPickupTravel = false
        state.restocking = false
        if not state.running or state.closed then return end
        local before = revision
        -- Finish the adapter-owned seed request before normal Snapshot/Prepare.
        if seedOverflow(now,false,before) then return end
        local view, waiting, blocked = read()
        if state.closed or not state.running or revision ~= before then return end
        if not view then
            release()
            if blocked then stop(waiting) else status("waiting", waiting) end
            return
        end
        local overheadCount, rowCount = blockCount(view), seedCount(view)
        if activeBlockCount and overheadCount ~= activeBlockCount then
            stop("Overhead layout changed. Press START to use the new block count.") return
        end
        if farmMode ~= "break_only" and activeSeedCount and rowCount ~= activeSeedCount then
            stop("Seed row layout changed. Press START to use the new seed count.") return
        end
        if activeRowKey and rowKey(view)~=activeRowKey then stop("Farm row changed. Press START at the new row.") return end
        local firstView = activeRowKey == nil
        activeBlockCount, activeSeedCount = overheadCount, rowCount
        activeRowKey=rowKey(view)
        counts(view)
        if firstView then
            if farmMode == "break_only" then state.batchPhase="blocks" end
            restoreProgress(view)
        end
        if not settle(view, now) then return end
        if suspendRequested and canSuspend() then release() return end
        if finite(view.nextReadyAt) then nextAction = math.max(nextAction, view.nextReadyAt) end
        local missing, plantingPending = 0, false
        if farmMode ~= "break_only" then
        missing,plantingPending=seedWork(view)
        -- Finish a block-drop receipt/cleanup before refilling or switching to
        -- tree work, even when maturity/stock changes during a partial sweep.
        local treeWorkNext = state.batchPhase=="growing" and (missing>0 or plantingPending or readyToHarvest(view))
            or state.batchPhase=="planting" and not plantingStarted and missing>0 and view.seeds>=missing
        if cleanupPending and not overheadDirty and treeWorkNext then
            -- An initial scan has no punch-derived receipt to protect. Do not
            -- let unrelated/stale drops delay a newly ready planting/harvest pass.
            release()
            if state.closed or not state.running or revision ~= before then return end
            cleanupPending = false
        end
        if cleanupPending or overheadDirty and treeWorkNext then
            if not cleanup(now,before) then return end
            if type(adapter.CleanupBlockDrops)=="function" then
                view=afterCleanup(now,before)
                if not view then return end
                missing,plantingPending=seedWork(view)
            end
        end
        local betweenBatches=state.batchPhase=="growing"
            or state.batchPhase=="planting" and (not plantingStarted or missing==0 and not plantingPending)
            or state.batchPhase=="harvesting" and state.occupied==0
        if suspendRequested and canSuspend() then release() return end
        if betweenBatches and next(pending)==nil and not cleanupPending and not overheadDirty
            and view.seedId==17 and view.blockId==16 and view.seeds>=180 then
            if seedOverflow(now,true,before) then return end
        end
        if not rowCursor then beginBatch(state.batchPhase,view) end
        if state.batchPhase=="growing" and (missing>0 or plantingPending) then beginBatch("planting",view) end
        if state.batchPhase=="harvesting" and state.occupied==0 then beginBatch("planting",view) end
        if state.batchPhase=="planting" then
            if missing==0 and not plantingPending then
                beginBatch("growing",view)
            elseif not plantingStarted and view.seeds<missing then
                -- Before a planting batch begins, build seed stock using the
                -- same centered overhead loop used while trees are growing.
                state.restocking=true
            else
                plantingStarted=true
                for _=1,rowCount do
                    if rowCursor<1 or rowCursor>rowCount then
                        beginBatch("planting",view)
                        status("waiting","Checking unfinished seed slots before the next planting pass.")
                        return
                    end
                    local i=rowCursor
                    if view.seedSlots[i].state=="empty" then
                        if isPending("Plant",i) then status("waiting","Waiting for planting confirmation.") return end
                        if view.seeds<1 then status("waiting","Need "..missing.." "..state.seedName.." to finish this planting batch.") release() return end
                        status("planting","Planting "..state.seedName.." "..i.."/"..rowCount.." · fixed sweep.")
                        if prepare("Plant",i,now,before) and now>=nextAction then send("Plant",i,now,before) end
                        return
                    end
                    rowCursor=rowCursor+rowDirection
                end
                return
            end
        end
        if suspendRequested and canSuspend() then release() return end
        if state.batchPhase=="growing" and readyToHarvest(view) then beginBatch("harvesting",view) end
        if state.batchPhase=="harvesting" then
            for _=1,rowCount do
                if rowCursor<1 or rowCursor>rowCount then rowCursor=rowDirection==1 and 1 or rowCount end
                local i=rowCursor
                if isPending("Harvest",i) then status("waiting","Waiting for harvest confirmation.") return end
                local slot=view.seedSlots[i]
                if slot.state=="mature" then
                    status("harvesting","Harvesting "..state.blockName.." tree "..i.."/"..rowCount.." · finish the row before replanting.")
                    if prepare("Harvest",i,now,before) and now>=nextAction then send("Harvest",i,now,before) end
                    return
                elseif slot.state=="growing" then
                    release()
                    status("waiting","Waiting for tree "..i.."/"..rowCount.." to mature before finishing this harvest batch.")
                    return
                end
                rowCursor=rowCursor+rowDirection
            end
            return
        end
        else
            if cleanupPending and not cleanup(now,before) then return end
        end
        if suspendRequested and canSuspend() then release() return end
        -- Tree work can take the character along the extended row. Return to the
        -- original block-farm center before overhead work or waiting for stock.
        if not prepare("Return", nil, now, before) then return end
        if initialCleanupNeeded and not cleanup(now,before) then return end
        if suspendRequested and canSuspend() then release() return end
        state.allowPickupTravel=not state.restocking
        local function overheadStatus(phase,message)
            if state.restocking then
                status("restocking","Need "..(missing-view.seeds).." more "..state.seedName.." before planting. "..message)
            elseif farmMode~="break_only" and state.mature==rowCount and grassStockGate(view) then
                status(phase,"Waiting for Grass stock <30 (bag: "..view.blocks.."). "..message)
            else status(phase,message) end
        end
        if mode == "fill" then
            local placementPending = false
            for i = 1, overheadCount do
                if view.blockSlots[i].state == "empty" then
                    if isPending("Place", i) then placementPending = true
                    elseif view.blocks > 0 then
                        overheadStatus("placing", "Placing "..state.blockName.." above the character " .. i .. "/" .. overheadCount .. ".")
                        if prepare("Place", i, now, before) and now >= nextAction then send("Place", i, now, before) end
                        return
                    end
                end
            end
            if placementPending then overheadStatus("waiting", "Waiting for block placement confirmation.") return end
            mode = "punch"
        end
        -- Each successful punch advances one overhead slot. A completed sweep
        -- refills missing blocks; a newly planted row starts overhead work at 1.
        for _ = 1, overheadCount do
            local i = cursor
            state.blockIndex = i
            if view.blockSlots[i].state == "grass" then
                local sequence = overheadCount == 3 and "1 → 2 → 3" or "1 → 2 → 3 → 4 → 5"
                overheadStatus("punching", "Punching "..state.blockName.." " .. i .. "/" .. overheadCount .. " · repeat " .. sequence .. ".")
                if prepare("PunchBlock", i, now, before) and now >= nextAction and send("PunchBlock", i, now, before) then
                    cursor = i == overheadCount and 1 or i + 1
                    state.blockIndex = cursor
                    if i == overheadCount then finishSweep() end
                end
                return
            end
            cursor = i == overheadCount and 1 or i + 1
            state.blockIndex = cursor
            if i == overheadCount then finishSweep() break end
        end
        overheadStatus("waiting", "Waiting for "..state.blockName..(farmMode == "break_only" and " blocks." or " blocks / seeds."))
    end
    function api.Start()
        if state.closed then return false end
        suspendRequested=false
        revision = revision + 1
        release()
        state.running = true
        state.allowPickupTravel=false
        state.restocking=false
        state.planted, state.placed, state.punches, state.harvested, state.blockIndex = 0, 0, 0, 0, 1
        nextAction, attempts = 0, {}
        status("starting", farmMode == "break_only" and "Preparing overhead blocks at the saved position."
            or "Preparing the selected seed row at the saved farm position.")
        local before = revision
        local view, waiting, blocked = read()
        if state.closed or not state.running or revision ~= before then return false end
        if view then
            local changed=not activeRowKey or rowKey(view)~=activeRowKey
                or farmMode ~= "break_only" and activeSeedCount~=seedCount(view)
            if changed then
                pending={}
                cleanupPending, initialCleanupNeeded, overheadDirty, sweepTouched = false, true, false, false
                if farmMode == "break_only" then state.batchPhase="blocks"
                else beginBatch("planting",view) end
            end
            if changed or activeBlockCount~=blockCount(view) then mode,cursor,state.blockIndex="fill",1,1 end
            if changed then restoreProgress(view) end
            activeBlockCount,activeSeedCount,activeRowKey=blockCount(view),seedCount(view),rowKey(view)
            counts(view)
        elseif blocked then stop(waiting)
        else status("waiting", waiting) end
        return state.running
    end
    function api.Stop()
        if not state.closed then stop("Stopped manually.") end
    end
    function api.CanSuspend()
        -- Finish native requests and the current overhead drop cleanup before
        -- handing movement/inventory to the shopping controller.
        return not stepping and canSuspend()
    end
    function api.RequestSuspend(value)
        suspendRequested=value==true
    end
    function api.ExportProgress()
        if not activeRowKey then return nil end
        return {
            version=1, world=world, mode=farmMode, rowKey=activeRowKey,
            seedCount=farmMode == "break_only" and 0 or activeSeedCount, blockCount=activeBlockCount,
            phase=state.batchPhase, rowDirection=rowDirection, rowCursor=rowCursor,
            plantingStarted=plantingStarted, blockCursor=cursor,
        }
    end
    function api.Step(now)
        if stepping or not finite(now) then return end
        stepping = true
        local ok, failure = pcall(step, now)
        stepping = false
        if not ok and not state.closed then stop("Farm paused: " .. tostring(failure):sub(1, 130)) end
    end
    function api.Close()
        if state.closed then return end
        stop("Closed.")
        state.closed = true
    end
    return api
end

end)()

local makePickupDriver = (function()
-- Short native walking/jumping trips along the configured farm, then back to center.
return function(adapter, options)
    options=options or {}
    local rowRadius=options.radius or 2
    assert(rowRadius==2 or rowRadius==10,"Unsupported pickup row radius.")
    local state = { busy = false, status = "Pickup idle.", failed = nil, targetUid = nil, returning = false }
    local api = { State = state }
    local anchorX, anchorY, routeStarted, targetStarted, targetUid
    local moveDirection, jumpHeld, jumpUntil, nextJump = 0, false, nil, 0
    local moveSince, moveX, cooldownUntil, stepping = nil, nil, 0, false
    local ignored, released, revision = {}, true, 0
    local function finite(n) return type(n) == "number" and n == n and n > -math.huge and n < math.huge end
    local function validUid(uid) return type(uid) == "string" and uid ~= "" or finite(uid) end
    local function releaseAll(force)
        if force or not released then
            if type(adapter.Release) == "function" then pcall(adapter.Release)
            else
                if type(adapter.Move) == "function" then pcall(adapter.Move, 0) end
                if type(adapter.Jump) == "function" then pcall(adapter.Jump, false) end
            end
        end
        moveDirection, jumpHeld, jumpUntil, moveSince, moveX, released = 0, false, nil, nil, nil, true
    end
    local function resetRoute()
        state.busy, state.returning, state.targetUid = false, false, nil
        anchorX, anchorY, routeStarted, targetStarted, targetUid = nil, nil, nil, nil, nil
    end
    local function cancel(message, force)
        revision = revision + 1
        releaseAll(force)
        resetRoute()
        state.failed, state.status = nil, message or "Pickup stopped."
    end
    local function fail(message)
        revision = revision + 1
        releaseAll(true)
        resetRoute()
        state.failed, state.status = message, message
    end
    local function move(direction)
        if direction ~= moveDirection then
            local before = revision
            adapter.Move(direction)
            if revision ~= before then return end
            moveDirection = direction
            if direction ~= 0 then released = false end
        end
        if direction == 0 then moveSince, moveX = nil, nil end
    end
    local function jump(down)
        if jumpHeld == down then return true end
        local before = revision
        adapter.Jump(down)
        if revision ~= before then return false end
        jumpHeld = down
        if down then released = false else jumpUntil = nil end
        return true
    end
    local function grounded(view)
        return view.grounded == true and math.abs(view.y - anchorY) < 0.15
    end
    local function eligible(drop, ax, ay)
        return type(drop) == "table" and validUid(drop.uid) and finite(drop.x) and finite(drop.y)
            and (drop.id == 16 or drop.id == 17 or drop.id == 112)
            and (drop.n == nil or finite(drop.n) and drop.n > 0)
            and math.abs(drop.x - ax) <= rowRadius+.9 and drop.y >= ay - 2 and drop.y <= ay + 1
            and not ignored[drop.uid]
    end
    local function dropsAt(view, ax, ay)
        local present, candidates = {}, {}
        for _, drop in pairs(type(view.drops) == "table" and view.drops or {}) do
            if type(drop) == "table" and validUid(drop.uid) then present[drop.uid] = true end
            if eligible(drop, ax, ay) then candidates[#candidates + 1] = drop end
        end
        -- A timed-out uid is ignored until it disappears, preventing endless jumps.
        for uid in pairs(ignored) do if not present[uid] then ignored[uid] = nil end end
        return candidates
    end
    local function choose(candidates, x, y)
        local best, distance
        for _, drop in ipairs(candidates) do
            local d = math.abs(drop.x - x) + math.abs(drop.y - y) * 0.2
            if not best or d < distance then best, distance = drop, d end
        end
        return best
    end
    local function walkToward(x, destination, now)
        local dx = destination - x
        if math.abs(dx) <= 0.3 then move(0) return true end
        local direction = dx < 0 and -1 or 1
        if not moveSince then moveSince, moveX = now, x
        elseif math.abs(x - moveX) > 0.15 then moveSince, moveX = now, x end
        if now - moveSince >= 8 then fail("Pickup path blocked for 8 seconds.") return false end
        move(direction)
        return false
    end
    local function finish(now)
        releaseAll(true)
        resetRoute()
        cooldownUntil = now + 0.8
        state.status = "Back at farm center."
    end
    local function step(now)
        if jumpHeld and jumpUntil and now >= jumpUntil then jump(false) end
        local before = revision
        local ok, view = pcall(adapter.View)
        if revision ~= before then return end
        if not ok or type(view) ~= "table" then cancel("Waiting for pickup data.") return end
        if view.enabled ~= true then cancel("Pickup stopped.") return end
        if view.ready ~= true then cancel("Waiting for the farm world.") return end
        if not finite(view.x) or not finite(view.y) or not finite(view.anchorX) or not finite(view.anchorY) then
            cancel("Waiting for valid pickup position.") return
        end
        if state.failed then return end
        if state.busy then
            if math.abs(view.anchorX - anchorX) > 0.001 or math.abs(view.anchorY - anchorY) > 0.001 then
                cancel("Farm anchor changed; pickup cancelled.") return
            end
            if math.abs(view.x - anchorX) > rowRadius+2 then fail("Pickup left the farm row.") return end
            if view.y - anchorY > .6 then fail("Pickup fell below the farm platform.") return end
            if now-routeStarted>=20 then fail("Pickup could not return to the farm platform.") return end
        end
        local candidates = dropsAt(view, state.busy and anchorX or view.anchorX, state.busy and anchorY or view.anchorY)
        if not state.busy then
            if now < cooldownUntil then return end
            if math.abs(view.x - view.anchorX) > 0.5 or view.grounded ~= true or math.abs(view.y - view.anchorY) >= 0.15 then
                state.status = "Stand at the farm center before pickup." return
            end
            if #candidates == 0 then state.status = "Pickup idle." return end
            anchorX, anchorY, routeStarted, nextJump = view.anchorX, view.anchorY, now, now
            state.busy, state.returning = true, false
        end
        if now - routeStarted >= 12 then state.returning = true end
        local target
        if not state.returning then
            if targetUid then
                for _, drop in ipairs(candidates) do if drop.uid == targetUid then target = drop break end end
                if target and now - targetStarted >= 3 then
                    ignored[targetUid], target = true, nil
                    local filtered = {}
                    for _, drop in ipairs(candidates) do if not ignored[drop.uid] then filtered[#filtered + 1] = drop end end
                    candidates = filtered
                end
            end
            if not target then
                target = choose(candidates, view.x, view.y)
                targetUid, targetStarted = target and target.uid or nil, target and now or nil
                state.targetUid = targetUid
            end
            if not target then state.returning = true end
        end
        if state.returning then
            targetUid, targetStarted, state.targetUid = nil, nil, nil
            state.status = "Returning to farm center."
            jump(false)
            local aligned = walkToward(view.x, anchorX, now)
            if not state.busy then return end
            if aligned and grounded(view) then finish(now) end
            return
        end
        state.status = "Picking up Grass / seeds / gems."
        local aligned = walkToward(view.x, target.x, now)
        if not state.busy then return end
        if aligned and target.y < view.y - 0.85 and grounded(view) and not jumpHeld and now >= nextJump then
            if jump(true) then jumpUntil, nextJump = now + 0.12, now + 0.8 end
        end
    end
    function api.Step(now)
        if stepping or not finite(now) then return end
        stepping = true
        local ok, failure = pcall(step, now)
        stepping = false
        if not ok then fail("Pickup stopped: " .. tostring(failure):sub(1, 110)) end
    end
    function api.Cancel()
        cancel("Pickup stopped.", true)
        cooldownUntil, ignored = 0, {}
    end
    return api
end

end)()

local makeAdapter = (function()
-- Native paired seed/block actions: a captured configurable row and three overhead slots.
return function(player, app)
    local R = game:GetService("ReplicatedStorage")
    local Input = game:GetService("UserInputService")

                    and type(rawget(value,"fg")) == "table" then ctx = value break end
            end

    end

                if name == "act" and u[6] == ctx then act, inventory, canPlace, setTile, plantSeed = fn, u[14], u[22], u[24], u[30] end

        end
    end
    assert(hit and act and type(canPlace) == "function" and type(plantSeed) == "function" and type(setTile) == "function"
        and type(treeRipe) == "function" and type(inventory) == "table" and type(inventory.count) == "function", "Native farm interface belum siap.")
    assert(debug.info(canPlace, "n") == "canPlace" and debug.info(plantSeed, "n") == "plantSeed"
        and debug.info(setTile, "n") == "setTile", "Native farm interface berubah.")
    local adapter = { Context = ctx, ReadyAt = 0, Recent = {}, Collected = 0, BlockCount = 3, SeedCount = 21, RowRadius = 10,
        CollectedBlocks = 0, CollectedSeeds = 0, CollectedGems = 0 }
    local visual=app.BackgroundOptimizer
    if visual and type(visual.BindFarmEffects)=="function" then
        -- These are presentation objects only. Leave the native plant/hit actions
        -- and their tile/inventory/network side effects untouched.
        pcall(function()
            local actValues=debug.getupvalues(act)
            local plantValues=debug.getupvalues(plantSeed)
            local pg=player:FindFirstChildOfClass("PlayerGui")
            local gui=pg and pg:FindFirstChild("GameGui")
            local particlesModule=gui and gui:FindFirstChild("Particles")
            if plantValues[4]==ctx and particlesModule and particlesModule:IsA("ModuleScript") then
                visual.BindFarmEffects({particles=actValues[23],particleClass=require(particlesModule),
                    plantFlights=plantValues[8],plantParent=plantValues[6]})
            end
        end)
    end
    local sessionJob = game.JobId
    local failedPickup,retryAt={},{}
    local PICKUP_FAILURE_LIMIT,PICKUP_DEFER_SECONDS,CLEANUP_BLOCK_SECONDS=3,15,8
    local blockClaims,breakWindows,firstSeen,rejectedClaims={},{},{},{}
    local provenanceBinding,provenanceEpoch=nil,0
    local function clearBlockProvenance()
        blockClaims,breakWindows,firstSeen,rejectedClaims={},{},{},{}
        failedPickup,retryAt={},{}
        provenanceBinding=nil
        provenanceEpoch=provenanceEpoch+1
    end
    local crops, cropBySeed = {}, {}
    local excluded = { "lock", "area", "door", "entrance", "portal", "checkpoint", "provider", "toggle", "spin",
        "dice", "vend", "hub", "display", "displayOne", "donation", "combiner", "hurts", "deadly", "bumper", "laser", "fruits" }
    -- Ordinary native tree fruit is seedId - 1. Family seeds override that with
    -- fruits / seedId aliases and are intentionally outside this one-pair flow.
    for id, seed in pairs(items) do
        if type(id)=="number" and id%1==0 and id>0 and type(seed)=="table" and seed.kind=="seed" and not seed.fruits then
            local block=rawget(items,id-1)
            local allowed=type(block)=="table" and block.farmable==true and not block.background and not block.kind
                and type(block.hits)=="number" and block.hits>0 and block.hits<math.huge
                and (block.seedId==nil or block.seedId==id)
            -- Normal Lava is an explicitly requested farm pair. It still uses
            -- the same native placement/range checks and stationary pickup.
            local normalLava=allowed and id==5 and block.name=="Lava" and seed.name=="Lava Seed"
            if allowed then
                for _,field in ipairs(excluded) do
                    if block[field] and not (normalLava and field=="hurts") then allowed=false break end
                end
            end
            if allowed then
                local crop={seedId=id,blockId=id-1,seedName=tostring(seed.name or id),blockName=tostring(block.name or id-1),
                    grow=type(seed.grow)=="number" and seed.grow or 0}
                crops[#crops+1]=crop cropBySeed[id]=crop
            end
        end
    end
    table.sort(crops,function(a,b)
        local an,bn=a.blockName:lower(),b.blockName:lower()
        return an==bn and a.seedId<b.seedId or an<bn
    end)

            local ownedA,ownedB=a.seeds>0 or a.blocks>0,b.seeds>0 or b.blocks>0
            if ownedA~=ownedB then return ownedA end
            local an,bn=a.blockName:lower(),b.blockName:lower()

    local function sameProvenance()
        local b=provenanceBinding
        local a=app.Anchor
        return b and a and b.anchor==a and b.world==ctx.name and b.job==game.JobId
            and b.x==a.x and b.y==a.y and b.mode==farmMode() and b.count==(a.plantCount or 21)

    local function refreshBlockProvenance(now)
        if provenanceBinding and not sameProvenance() then clearBlockProvenance() end
        if not provenanceBinding then return end
        local live={}
        for _,d in pairs(ctx.dropList()) do
            if type(d)=="table" and d.uid~=nil then
                live[d.uid]=d
                if not firstSeen[d.uid] then
                    firstSeen[d.uid]={object=d,id=d.id,n=d.n,x=d.x,y=d.y,born=d.born}
                end
                local claim=blockClaims[d.uid]
                if claim then
                    if claim.object~=d or claim.id~=d.id or type(d.n)~="number" or d.n>claim.lastN or d.pending or d.display then
                        blockClaims[d.uid]=nil rejectedClaims[d.uid]=true
                        failedPickup[d.uid],retryAt[d.uid]=nil,nil
                    else claim.lastN=d.n end
                end
            end
        end
        for uid in pairs(blockClaims) do if not live[uid] then
            blockClaims[uid]=nil failedPickup[uid],retryAt[uid]=nil,nil
        end end
        for uid in pairs(firstSeen) do if not live[uid] then firstSeen[uid]=nil end end
        for uid in pairs(rejectedClaims) do if not live[uid] then rejectedClaims[uid]=nil end end
        for i=#breakWindows,1,-1 do
            local window=breakWindows[i]
            if now>window.expires then table.remove(breakWindows,i)
            else
                for uid,d in pairs(live) do
                    local born=firstSeen[uid]
                    if not blockClaims[uid] and not rejectedClaims[uid] and not window.baseline[uid]
                        and born.object==d and born.id==d.id and born.n==d.n
                        and (d.id==window.blockId or d.id==window.seedId or d.id==112)
                        and type(d.n)=="number" and d.n>0 and not d.pending and not d.display
                        and type(born.born)=="number" and born.born>=window.started and born.born<=window.expires
                        and type(born.x)=="number" and type(born.y)=="number"
                        and math.abs(born.x-(window.x+.5))<=.75 and math.abs(born.y-(window.y+.5))<=.4 then
                        blockClaims[uid]={object=d,id=d.id,lastN=d.n}
                    end
                end
            end
        end
    end

    local virtualInput,held=game:GetService("VirtualInputManager"),{}

    local function key(code,down)
        if held[code]==down or (not down and not held[code]) then return end
        -- Volt callbacks can lose the input capability after native game calls.
        -- Restore the executor's input context only for this call, then restore it.
        local previous
        if type(getIdentity)=="function" then local ok,value=pcall(getIdentity) if ok then previous=value end end
        if previous~=nil and type(setIdentity)=="function" then pcall(setIdentity,8) end
        local sent,failure=pcall(function() virtualInput:SendKeyEvent(down,code,false,game) end)
        if previous~=nil and type(setIdentity)=="function" then pcall(setIdentity,previous) end
        if not sent then error("Kontrol keyboard Volt belum siap: "..tostring(failure)) end
        held[code]=down or nil
    end

        for _,code in ipairs({Enum.KeyCode.Left,Enum.KeyCode.Right,Enum.KeyCode.Up,Enum.KeyCode.Space}) do pcall(key,code,false) end

    end
    local function withinFarm()

    local function directPickupRange(d,x,y)

        return math.abs(d.x-x)<=1.3 and math.abs(d.y-(y-.475))<=1.3
    end
    local function centralDrop(d,a)
        return type(d.x)=="number" and type(d.y)=="number" and not d.display
            and math.abs(d.x-(a.x+.5))<=2.25 and d.y>=a.y-1.35 and d.y<=a.y+1.05
    end
    local function dropInFarmScope(d,a)
        if farmMode()=="break_only" then return centralDrop(d,a) end
        if centralDrop(d,a) then return true end
        local layout=adapter.GetPlantLayout()
        return layout and d.x>=a.x+.5-layout.left-.9 and d.x<=a.x+.5+layout.right+.9
            and math.abs(d.y-(a.y+.5))<=2 or false
    end
    local function centralDrops()

        refreshBlockProvenance(os.clock())
        local a=app.Anchor
        for _,d in pairs(ctx.dropList()) do
            if type(d)=="table" and blockClaims[d.uid] and blockClaims[d.uid].object==d
                and type(d.n)=="number" and d.n>0 and centralDrop(d,a) then
                result[#result+1]=d
                if d.id~=112 and inventory:room(d.id)<=0 then full=full+1
                elseif blockClaims[d.uid].blockingExhausted then deferred=deferred+1
                else available=available+1 end
            end
        end
        return result,available,full,deferred
    end

    local function breakOnlyLayoutIssue(a)
        local tr=trees()
        if type(tr)~="table" then return "Menunggu data area tiga blok." end
        for x=a.x-1,a.x+1 do
            local support=items[fg(x,a.y+1)]
            if not support or (support.solid~=true and support.platform~=true)
                or support.hurts or support.deadly or support.bumper or support.laser then
                return "Break Only memerlukan lantai aman di tiga petak tengah."
            end
            if not permitted(x,a.y,false) or not permitted(x,a.y-1,false) then
                return "Tiga petak tengah tidak memiliki izin farm."
            end
            local id=fg(x,a.y-1)
            if tr[(a.y-1)*ctx.w+x] or (id~=0 and id~=blockId) then
                return "Petak atas berisi benda selain "..selectedCrop.blockName.."."
            end
            if id==0 and not permitted(x,a.y-1,true) then return "Petak atas tidak memiliki izin memasang blok." end
        end
        local body=items[fg(a.x,a.y)]
        if body and (body.solid or body.hurts or body.deadly or body.door or body.entrance or body.portal) then
            return "Berdiri di ruang kosong yang aman untuk Break Only."
        end
    end
    function adapter.Capture()
        clearBlockProvenance()
        adapter.Release()
        local issue = worldIssue() or paused()
        if issue then return nil, issue end
        local x,y = position()
        local mode=farmMode()
        local layout=adapter.GetPlantLayout()
        local a = {world=ctx.name, x=math.floor(x), y=math.floor(y-.475), job=game.JobId,mode=mode,
            plantCount=layout and layout.count or 21,seedId=seedId,blockId=blockId}
        local left=mode=="break_only" and 1 or math.max(layout.left,1)
        local right=mode=="break_only" and 1 or math.max(layout.right,1)
        if a.x < left or a.x+right >= ctx.w or a.y < 1 or a.y+1 >= ctx.h then
            return nil,mode=="break_only" and "Pilih titik dengan tiga petak atas di dalam world."
                or "Baris "..layout.count.." tanaman memerlukan "..layout.left.." petak kiri dan "..layout.right.." petak kanan, serta tiga blok atas di dalam world."
        end
        if not permitted(a.x,a.y,true) then return nil, "Titik farm tidak memiliki izin membangun." end
        if mode=="break_only" then
            local layout=breakOnlyLayoutIssue(a)
            if layout then return nil,layout end
            if math.abs(y-(a.y+1))>.12 then return nil,"Mendarat di lantai dahulu sebelum SET TITIK Break Only." end
        end
        return a
    end
    function adapter.Inventory() return math.max(0,inventory:count(seedId)), math.max(0,inventory:count(blockId)) end
    function adapter.Snapshot()
        local mode=farmMode()
        local layout=adapter.GetPlantLayout()
        local s = {ready=false, world=ctx.name, mode=mode,atAnchor=false, canFarm=false, blockCount=3, seedCount=mode=="break_only" and 0 or (layout and layout.count or 0),
            canTravel=true, allowPartialPlanting=true, seedSlots={}, blockSlots={},seedId=seedId,blockId=blockId,
            seedName=selectedCrop.seedName,blockName=selectedCrop.blockName,crop=adapter.GetCrop(),
            rowLeft=mode=="break_only" and 0 or (layout and layout.left or 0),rowRight=mode=="break_only" and 0 or (layout and layout.right or 0)}
        s.seeds,s.blocks = adapter.Inventory()
        local issue = anchorIssue()
        if issue then s.message=issue s.invalidWorld=worldIssue()~=nil or app.Anchor~=nil return s end
        local drops,available,full,deferred=centralDrops()
        s.centralDrops,s.centralCollectible,s.centralCapacityBlocked,s.centralDeferred=#drops,available,full,deferred
        s.centralCleanupStatus=adapter.LastBlockCleanup
        local pause = paused()
        if pause then s.message=pause return s end
        if app.Pickup and app.Pickup.State.busy then s.message=app.Pickup.State.status return s end
        s.ready=true
        s.canFarm=true
        s.atAnchor=atAnchor()
        if not withinFarm() then s.canFarm=false s.message="Karakter keluar dari baris farm. STOP lalu kembali ke titik." return s end
        local tr = trees()
        if type(tr) ~= "table" then s.ready=false s.message="Menunggu data tanaman." return s end
        if mode=="break_only" then
            local layout=breakOnlyLayoutIssue(app.Anchor)
            if layout then s.canFarm=false s.message=layout return s end
        end
        s.canFarm=true
        local px,py=position()
        for i=1,s.seedCount do
            local x,y=coords(i,false)
            local f,t=fg(x,y),tr[y*ctx.w+x]
            local slot={state="blocked", x=x,y=y,distance=math.abs(x+.5-px)}
            local support=items[fg(x,y+1)]
            local floorOK=support and (support.solid==true or support.platform==true)
            if permitted(x,y,false) then
                if f==0 and not t then
                    if floorOK and permitted(x,y,true) then slot.state="empty" else slot.message="Tanah/ruang tanam belum sesuai." end
                elseif f==seedId then
                    slot.state="growing"
                    if t and type(t.p)=="number" then
                        slot.remaining=math.max(0,t.p+selectedCrop.grow-workspace:GetServerTimeNow())
                        if treeRipe(seedId,t) then slot.state="mature" end
                    end
                else slot.message="Petak tanam berisi benda selain "..selectedCrop.seedName.."." end
            else slot.message="Petak tanam tidak dapat digunakan." end
            if not floorOK then slot.state="blocked" slot.message="Tanah penopang baris farm terputus." end
            s.seedSlots[i]=slot
            if slot.state=="blocked" then s.message=slot.message .. " Slot " .. i .. "." end
        end
        for i=1,3 do
            local bx,by=coords(i,true)
            local bf=fg(bx,by)
            local b={state="blocked",x=bx,y=by}
            if permitted(bx,by,false) and not tr[by*ctx.w+bx] then
                if bf==blockId then b.state="grass"
                elseif bf==0 and permitted(bx,by,true) then b.state="empty"
                else b.message="Petak atas berisi benda selain "..selectedCrop.blockName.."." end
            else b.message="Petak atas tidak dapat digunakan." end
            s.blockSlots[i]=b
            if b.state=="blocked" then s.message=b.message .. " Slot " .. i .. "." end
        end
        s.nextReadyAt=math.max(adapter.ReadyAt,ctx.punchReady or 0)
        return s
    end
    function adapter.Prepare(kind,i,now)
        if not app.Alive or not app.Engine or not app.Engine.State.running then adapter.Release() return false,"Farm berhenti." end
        if farmMode()=="break_only" and (kind=="Plant" or kind=="Harvest") then
            adapter.Release() return false,"Break Only tidak menanam atau memanen seed.",true
        end
        if paused() then adapter.Release() return false,paused() end
        if not withinFarm() then adapter.Release() return false,"Karakter keluar dari baris farm.",true end
        if app.Pickup and app.Pickup.State.busy then return false,app.Pickup.State.status end
        local px,py=position()
        local destination,tolerance
        if kind=="Plant" or kind=="Harvest" then
            local x,y=coords(i,false)
            if not x or not permitted(x,y,kind=="Plant") then adapter.Release() return false,"Petak tanaman tidak dapat digunakan.",true end
            destination=x+.5
            local range=kind=="Plant" and helpers.buildRange() or helpers.punchRange()
            tolerance=math.min(kind=="Plant" and 1.6 or .75,range-.15)
        else
            destination=app.Anchor.x+.5
            tolerance=.2
        end
        if py-(app.Anchor.y+1)>.6 then adapter.Release() return false,"Karakter jatuh dari baris farm.",true end
        if math.abs(py-(app.Anchor.y+1))>.12 then adapter.Move(0) return false,"Menunggu karakter mendarat di baris farm." end
        local dx=destination-px
        if math.abs(dx)<=tolerance then releaseMovement() return true end
        local targetKey=kind..":"..tostring(i or "center")
        if not navigation or navigation.target~=targetKey then navigation={target=targetKey,at=now,x=px,started=now}
        elseif math.abs(px-navigation.x)>.15 then navigation.x,navigation.at=px,now end
        if now-navigation.at>8 or now-navigation.started>25 then adapter.Release() return false,"Jalur tanam/panen terhalang; periksa baris farm.",true end
        adapter.TravelBusy=true
        adapter.Move(dx<0 and -1 or 1)
        local count=adapter.GetPlantLayout()
        return false,(kind=="Plant" and "Menuju petak tanam "..i.."/"..count.count.."." or (kind=="Harvest" and "Menuju tanaman matang "..i.."/"..count.count.."." or "Kembali ke tengah untuk farm tiga blok."))
    end
    function adapter.RecoveryView()
        local issue=anchorIssue()
        if issue then return {ready=false,message=issue,fatal=true,world=ctx.name,job=game.JobId} end
        local pause=paused()
        if pause then return {ready=false,message=pause,world=ctx.name,job=game.JobId} end
        local x,y=position()
        return {ready=true,world=ctx.name,job=game.JobId,x=x,feetY=y,anchorX=app.Anchor.x+.5,
            anchorFeetY=app.Anchor.y+1,mode=farmMode(),plantCount=farmMode()=="break_only" and 0 or adapter.GetPlantLayout().count}
    end
    local function recoveryBody(x,y,startX,startY)
        if not inside(x,y) or not permitted(x,y,false) then return false end
        local id=fg(x,y)
        local item=items[id]
        if id==nil or (id~=0 and not item) then return false end
        if item then
            if item.hurts or item.deadly or item.bumper or item.laser or item.solid then return false end
            if (item.door or item.entrance or item.portal) and not (x==startX and y==startY) then return false end
        end
        return true
    end
    local function recoveryFloor(x,y)
        local item=items[fg(x,y+1)]
        return item and (item.solid==true or item.platform==true) and not (item.hurts or item.deadly or item.bumper
            or item.laser or item.door or item.entrance or item.portal or item.lock) or false
    end
    local function overflowStatus(message)
        adapter.SeedOverflowStatus,adapter.OverflowStatus=message,message
        return message
    end
    local function overflowBinding()
        local a=app.Anchor
        local layout=adapter.GetPlantLayout()
        return {anchor=a,x=a.x,y=a.y,world=ctx.name,job=game.JobId,count=layout.count,
            seedId=seedId,blockId=blockId,revision=overflowRevision}
    end
    local function overflowCurrent(bound)
        local a=app.Anchor
        local layout=adapter.GetPlantLayout()
        return bound and app.Alive and app.Engine and app.Engine.State.running
            and bound.revision==overflowRevision
            and bound.anchor==a and a.x==bound.x and a.y==bound.y
            and ctx.name==bound.world and app.World==bound.world and game.JobId==bound.job
            and layout and layout.count==bound.count and seedId==bound.seedId and blockId==bound.blockId
            and farmMode()=="farm" and seedId==17 and blockId==16 and anchorIssue()==nil
    end
    local function validRecycleCount(value)
        return type(value)=="number" and value==value and value>=0 and value<math.huge and value%1==0
    end
    local function nativeRecycler()
        local fn=rawget(helpers,"onRecycle")
        if type(fn)~="function" or type(inventory.remove)~="function" then return end
        local ok,up=pcall(debug.getupvalues,fn)
        -- The native Game callback closes over this exact world context.
        if ok and type(up)=="table" and up[1]==ctx then return fn end
    end
    local function finishRecycle(token,cancelled)
        token.nativePending=false token.done=true token.cancelled=cancelled==true
        token.completedAt=os.clock()
    end
    function adapter.ServiceSeedOverflow(now,allowStart)
        if type(now)~="number" or now~=now then now=os.clock() end
        if not app.Alive or not app.Engine or not app.Engine.State.running then
            if seedOverflow then adapter.Release() end
            return false
        end
        local pending=app.SeedOverflowPending
        if type(pending)=="table" then
            if pending.userId~=nil and pending.userId~=player.UserId then
                app.SeedOverflowPending=nil
            elseif pending.uncertain==true then
                -- A throwing native call might already have reached the server.
                -- Resume ordinary farming but never replay this recycle request.
                if seedOverflow or adapter.SeedOverflowActive then releaseMovement() end
                seedOverflow=nil adapter.SeedOverflowActive=false
                return false,overflowStatus("Recycle seed ditunda: hasil permintaan sebelumnya belum pasti; tidak dikirim ulang.")
            elseif pending.nativePending==true and pending.done~=true then
                releaseMovement() adapter.SeedOverflowActive=true
                if seedOverflow~=pending then seedOverflow=nil end
                return true,overflowStatus(pending.action=="recycle"
                    and "Menunggu recycle Grass Seed; tidak mengirim ulang."
                    or "Menunggu permintaan seed sebelumnya selesai; recycle belum dikirim.")
            else
                -- Completion is shared by reference across STOP/reload. Do not
                -- discard a submitted operation merely because world/job changed.
                if app.SeedOverflowPending==pending then app.SeedOverflowPending=nil end
                seedOverflow=nil adapter.SeedOverflowActive=false
                nextOverflowAt=math.max(nextOverflowAt,now+15)
                if pending.action=="recycle" and not pending.cancelled then
                    adapter.LastSeedOverflow={action="recycle",status="dispatched",requested=pending.requested,
                        before=pending.before,after=pending.after,at=pending.completedAt,
                        serverConfirmed=false}
                    return false,overflowStatus("Permintaan recycle Grass Seed dikirim; sisakan 21. Menunggu sinkronisasi game.")
                end
                return false,overflowStatus("Permintaan seed selesai / dibatalkan; farm dilanjutkan.")
            end
        end
        if allowStart~=true or farmMode()~="farm" or seedId~=17 or blockId~=16 or anchorIssue()
            or now<nextOverflowAt or paused() or collecting or (app.Pickup and app.Pickup.State.busy) then return false end
        local count=inventory:count(17)
        if not validRecycleCount(count) or count<180 then return false end
        local recycle=nativeRecycler()
        if not recycle then
            nextOverflowAt=now+15
            return false,overflowStatus("Recycle seed ditunda: antarmuka native belum tersedia.")
        end
        local bound=overflowBinding()
        if not overflowCurrent(bound) or paused() then return false end
        refreshBlockProvenance(now)
        releaseMovement()
        local token={action="recycle",bound=bound,userId=player.UserId,world=ctx.name,job=game.JobId,
            phase="queued",nativePending=true,done=false,started=now}
        seedOverflow=token app.SeedOverflowPending=token adapter.SeedOverflowActive=true
        local function dispatch()
            if seedOverflow~=token or app.SeedOverflowPending~=token or not overflowCurrent(bound) or paused()
                or nativeRecycler()~=recycle then finishRecycle(token,true) return end
            -- Match InventoryUI: clamp to the *current* stack, remove that
            -- amount locally once, then dispatch onRecycle(id, quantity) once.
            -- onRecycle discards the server reply; only InventorySync is
            -- authoritative. A normal nil return means dispatched, not confirmed.
            local currentCount=inventory:count(17)
            if not validRecycleCount(currentCount) or currentCount<180
                or seedOverflow~=token or app.SeedOverflowPending~=token or not overflowCurrent(bound) or paused()
                or nativeRecycler()~=recycle then finishRecycle(token,true) return end
            token.before=currentCount token.requested=currentCount-21
            token.phase="dispatching"
            -- No additional wait/read occurs between final validation and the
            -- native UI's debit/send pair. Do not independently refund or debit
            -- again after a late response; InventorySync owns reconciliation.
            local sent,problem=pcall(function()
                inventory:remove(17,token.requested)
                token.debited=true
                recycle(17,token.requested)
            end)
            if not sent then
                token.uncertain=true token.error=tostring(problem):sub(1,140)
                token.phase="uncertain"
                finishRecycle(token,false)
                return
            end
            token.phase="dispatched"
            local counted,remaining=pcall(inventory.count,inventory,17)
            if counted and validRecycleCount(remaining) then token.after=remaining end
            finishRecycle(token,false)
            -- STOP/close/reload can happen during the native call. The shared
            -- token is the only completion write: no movement, farm or retry.
        end
        local queued,queueError=pcall(task.spawn,function()
            local called,problem=pcall(dispatch)
            if not called then
                token.error=tostring(problem):sub(1,140)
                if token.phase=="queued" then finishRecycle(token,true)
                else token.uncertain=true finishRecycle(token,false) end
            end
        end)
        if not queued then
            token.error=tostring(queueError):sub(1,140)
            if token.phase=="queued" then finishRecycle(token,true)
            else token.uncertain=true finishRecycle(token,false) end
        end
        return true,overflowStatus("Recycle surplus Grass Seed; mempertahankan 21 seed di tas.")
    end
    local function validateRestoredAnchor(candidate)
        local function integer(n) return type(n)=="number" and n==n and n>-math.huge and n<math.huge and n%1==0 end
        if type(candidate)~="table" or not integer(candidate.x) or not integer(candidate.y) then
            return nil,"Koordinat titik tersimpan tidak valid.",true
        end
        local mode=farmMode()
        local layout,layoutIssue=adapter.GetPlantLayout()
        if mode~="farm" and mode~="break_only" or not layout then return nil,layoutIssue or "Mode farm tidak valid.",true end
        if type(candidate.world)~="string" or candidate.world=="" or candidate.world~=app.World then
            return nil,"World titik tersimpan berbeda dari world tujuan.",true
        end
        if type(ctx.name)~="string" or ctx.name=="" then return nil,"Menunggu identitas world.",false end
        if candidate.world~=ctx.name then return nil,"Titik tersimpan berasal dari world lain.",true end
        local savedMode=candidate.mode==nil and "farm" or candidate.mode
        local savedCount=candidate.plantCount==nil and 21 or candidate.plantCount
        if savedMode~=mode or not integer(savedCount) or savedCount~=layout.count then
            return nil,"Mode atau jumlah tanaman titik tersimpan berbeda.",true
        end
        if candidate.seedId~=seedId or candidate.blockId~=blockId then
            return nil,"Jenis seed/block titik tersimpan berbeda dari pilihan farm.",true
        end
        if not app.Alive or game.JobId~=sessionJob or app.Job~=nil and app.Job~=game.JobId then
            return nil,"Sesi farm berubah; muat ulang data world sebelum memulihkan titik.",true
        end
        if not integer(ctx.w) or not integer(ctx.h) or ctx.w<3 or ctx.h<3 then
            return nil,"Menunggu ukuran world termuat.",false
        end
        if type(ctx.lock)~="table" or not integer(ctx.lock.owner) or ctx.lock.owner<0 or type(ctx.areas)~="table"
            or type(ctx.areas.mayBuild)~="function" or type(ctx.builderOk)~="function" or type(areaLock.hasAccess)~="function" then
            return nil,"Menunggu data lock dan izin world.",false
        end
        local issue=worldIssue()
        if issue then return nil,issue,true end
        local left=mode=="break_only" and 1 or math.max(layout.left,1)
        local right=mode=="break_only" and 1 or math.max(layout.right,1)
        local ax,ay=candidate.x,candidate.y
        if ax-left<0 or ax+right>=ctx.w or ay<1 or ay+1>=ctx.h then
            return nil,"Baris tanaman atau tiga petak atas titik tersimpan berada di luar world.",true
        end
        local tr=trees()
        if type(tr)~="table" then return nil,"Menunggu data tanaman world.",false end
        local function loaded(x,y)
            local id=fg(x,y)
            return integer(id) and id>=0 and (id==0 or type(items[id])=="table")
        end
        -- Confirm hydration before interpreting air/unknown tiles as a missing
        -- floor or incompatible crop. No character position/input is touched.
        for x=ax-left,ax+right do
            if not loaded(x,ay) or not loaded(x,ay+1) then return nil,"Menunggu petak baris farm termuat.",false end
        end
        for x=ax-1,ax+1 do
            if not loaded(x,ay-1) then return nil,"Menunggu tiga petak block atas termuat.",false end
        end
        for x=ax-left,ax+right do
            if not recoveryFloor(x,ay) then return nil,"Lantai titik/baris farm tersimpan tidak aman atau terputus.",true end
        end
        if not recoveryBody(ax,ay) then return nil,"Ruang berdiri titik tersimpan tidak aman atau tidak diizinkan.",true end
        if mode=="farm" then
            for x=ax-layout.left,ax+layout.right do
                local id,tree=fg(x,ay),tr[ay*ctx.w+x]
                if not permitted(x,ay,false) then return nil,"Izin salah satu petak tanaman telah berubah.",true end
                if id==seedId then
                    if type(tree)~="table" or type(tree.p)~="number" or tree.p~=tree.p or math.abs(tree.p)==math.huge then
                        return nil,"Menunggu data seed/tanaman tersimpan lengkap.",false
                    end
                elseif id==0 then
                    if tree~=nil then return nil,"Menunggu sinkronisasi petak tanaman kosong.",false end
                    if not permitted(x,ay,true) then return nil,"Izin menanam salah satu petak telah berubah.",true end
                else return nil,"Baris tersimpan berisi tanaman/block selain pilihan farm.",true end
            end
        end
        for x=ax-1,ax+1 do
            local id,tree=fg(x,ay-1),tr[(ay-1)*ctx.w+x]
            if not permitted(x,ay,false) or not permitted(x,ay-1,false) then
                return nil,"Tiga petak tengah tidak lagi memiliki izin farm.",true
            end
            if tree~=nil then return nil,"Petak block atas berisi tanaman lain.",true end
            if id~=0 and id~=blockId then return nil,"Petak block atas berisi block selain pilihan farm.",true end
            if id==0 and not permitted(x,ay-1,true) then return nil,"Izin memasang block atas telah berubah.",true end
        end
        -- A native permission callback may yield. Recheck the binding before
        -- committing a fresh anchor; saved JobIds are deliberately not reused.
        if not app.Alive or game.JobId~=sessionJob or app.Job~=nil and app.Job~=game.JobId
            or ctx.name~=candidate.world or app.World~=candidate.world
            or farmMode()~=mode or seedId~=candidate.seedId or blockId~=candidate.blockId
            or not adapter.GetPlantLayout() or adapter.GetPlantLayout().count~=layout.count then
            return nil,"World atau konfigurasi berubah saat memvalidasi titik.",true
        end
        local finalIssue=worldIssue()
        if finalIssue then return nil,finalIssue,true end
        return {world=candidate.world,x=ax,y=ay,job=game.JobId,mode=mode,plantCount=layout.count,
            seedId=seedId,blockId=blockId},"Titik farm tersimpan telah divalidasi."
    end
    function adapter.RestoreAnchor(candidate)
        local ok,restored,message,fatal=pcall(validateRestoredAnchor,candidate)
        if not ok then return false,"Menunggu data native untuk memulihkan titik: "..tostring(restored):sub(1,110),false end
        if not restored then return false,message,fatal==true end
        app.Anchor=restored
        return true,message
    end
    local function recoveryJumpClear(fromX,fromY,toX,toY,startX,startY)
        if math.abs(toX-fromX)~=1 or fromY-toY<1 or fromY-toY>2 or not recoveryFloor(toX,toY) then return false end
        -- Native vertical takeoff first; adjacent movement starts only above
        -- the ledge. One extra clear head cell leaves room for the jump apex.
        for y=toY-1,fromY do
            if not recoveryBody(fromX,y,startX,startY) then return false end
        end
        return recoveryBody(toX,toY) and recoveryBody(toX,toY-1)
    end
    local function recoveryPlan(startX,startY,a)
        if not recoveryBody(startX,startY,startX,startY) or not recoveryFloor(startX,startY) then return nil end
        if not recoveryBody(a.x,a.y) or not recoveryFloor(a.x,a.y) then return nil end
        local queue,cursor,parents={{x=startX,y=startY}},1,{}
        local startKey=startY*ctx.w+startX
        parents[startKey]=false
        local endKey
        while queue[cursor] and cursor<=8000 do
            local node=queue[cursor]
            if node.x==a.x and node.y==a.y then endKey=node.y*ctx.w+node.x break end
            cursor=cursor+1
            local first=a.x>=node.x and 1 or -1
            for _,dir in ipairs({first,-first}) do
                local x,y=node.x+dir,node.y
                if recoveryBody(x,y,startX,startY) then
                    local landing
                    if recoveryFloor(x,y) then landing=y
                    elseif fg(x,y+1)==0 or recoveryBody(x,y+1,startX,startY) then
                        -- Only ordinary drops of up to four tiles. Never
                        -- break a floor or pass through a solid/hazard.
                        for below=y+1,math.min(y+4,ctx.h-2) do
                            if not recoveryBody(x,below,startX,startY) then break end
                            if recoveryFloor(x,below) then landing=below break end
                            local support=items[fg(x,below+1)]
                            if support and (support.solid or support.platform) then break end
                        end
                    end
                    if landing then
                        local id=landing*ctx.w+x
                        if parents[id]==nil and id~=startKey then
                            parents[id]={from=node.y*ctx.w+node.x,x=x,y=landing,fromX=node.x,fromY=node.y}
                            queue[#queue+1]={x=x,y=landing}
                        end
                    end
                end
                for rise=1,2 do
                    local landing=y-rise
                    local id=landing*ctx.w+x
                    if parents[id]==nil and id~=startKey and recoveryJumpClear(node.x,y,x,landing,startX,startY) then
                        parents[id]={from=node.y*ctx.w+node.x,x=x,y=landing,fromX=node.x,fromY=node.y,kind="jump"}
                        queue[#queue+1]={x=x,y=landing}
                    end
                end
            end
        end
        if not endKey then return nil end
        local reversed,path={},{}
        while endKey~=startKey do
            local edge=parents[endKey]
            if not edge then return nil end
            reversed[#reversed+1]=edge
            endKey=edge.from
        end
        for i=#reversed,1,-1 do path[#path+1]=reversed[i] end
        return path
    end
    function adapter.ReturnToAnchor(now)
        local function wait(message,fatal)
            adapter.RecoveryStatus=message
            return false,message,fatal
        end
        if not app.Alive or app.Recovering~=true or app.WantsRun~=true then
            adapter.Release() return wait("Perjalanan kembali dihentikan.")
        end
        local view=adapter.RecoveryView()
        if not view.ready then
            if view.fatal then adapter.Release() else releaseMovement() end
            return wait(view.message,view.fatal)
        end
        if type(now)~="number" or now~=now then now=os.clock() end
        local a=app.Anchor
        if recovery and (recovery.anchor~=a or recovery.x~=a.x or recovery.y~=a.y) then
            adapter.Release() return wait("Titik tujuan berubah; START ulang untuk kembali.",true)
        end
        if not recovery then recovery={anchor=a,x=a.x,y=a.y,nextPlan=now} end
        local state=recovery
        local x,feet=view.x,view.feetY
        local grounded=math.abs(feet-math.floor(feet+.5))<=.12
        local row=math.floor(feet+.5)-1
        if grounded and row==a.y and math.abs(x-view.anchorX)<=.2 then
            releaseMovement()
            if not recoveryBody(a.x,a.y) or not recoveryFloor(a.x,a.y) then
                return wait("Titik farm belum aman untuk ditempati; menunggu lantai/jalur siap.")
            end
            recovery=nil
            adapter.RecoveryStatus="Sudah kembali ke titik farm tersimpan."
            return true,adapter.RecoveryStatus
        end
        if not state.path then
            releaseMovement()
            if not grounded then return wait("Menunggu mendarat untuk kembali ke titik tersimpan.") end
            if now<state.nextPlan then return wait(state.blocked or "Menunggu jalur kembali tersedia.") end
            state.nextPlan=now+1
            local path=recoveryPlan(math.floor(x),row,a)
            if not path then
                state.blocked="Jalur kembali belum tersedia: perlu jalan/lantai aman atau undakan yang dapat dilompati normal."
                return wait(state.blocked)
            end
            -- A same-cell start can still need centering inside that cell.
            if #path==0 then path[1]={x=a.x,y=a.y,fromX=a.x,fromY=a.y} end
            state.path,state.index=path,1
            state.startX,state.startY=math.floor(x),row
            state.progressX,state.progressY,state.progressAt,state.legAt=x,feet,now,now
        end
        local edge=state.path[state.index]
        if edge.kind=="jump" then
            local jumpKey=edge.fromX..":"..edge.fromY..":"..edge.x..":"..edge.y
            state.jumpAttempts=state.jumpAttempts or {}
            local canJump=false
            if type(app.Session)=="table" and type(app.Session.View)=="function" then
                local ok,session=pcall(app.Session.View)
                canJump=ok and type(session)=="table" and session.canSpaceJump==true
            end
            if not canJump then
                releaseMovement()
                state.jump=nil
                return wait("Lompat Space belum tersedia pada kontrol native; menunggu pengaturan lompat.")
            end
            if app.Recovering~=true or app.WantsRun~=true or anchorIssue() or paused() then
                adapter.Release() return wait("Perjalanan kembali dihentikan.")
            end
            local column=math.floor(x)
            if not recoveryJumpClear(edge.fromX,edge.fromY,edge.x,edge.y,state.startX,state.startY)
                or (column~=edge.fromX and column~=edge.x) then
                releaseMovement()
                state.jump=nil state.path=nil state.nextPlan=now+1
                return wait("Ruang lompat berubah; menunggu jalur kembali yang aman.")
            end
            if grounded and row==edge.y and column==edge.x and math.abs(x-(edge.x+.5))<=.2 then
                releaseMovement()
                state.jumpAttempts[jumpKey]=nil
                state.jump=nil state.index=state.index+1
                if not state.path[state.index] then state.path=nil state.nextPlan=now end
                state.progressX,state.progressY,state.progressAt,state.legAt=x,feet,now,now
                return wait("Undakan terlewati; melanjutkan jalan ke titik farm.")
            end
            if not state.jump then
                if not grounded or row~=edge.fromY or column~=edge.fromX or not recoveryFloor(edge.fromX,edge.fromY) then
                    releaseMovement() state.path=nil state.nextPlan=now+1
                    return wait("Menunggu posisi pijakan siap sebelum lompat normal.")
                end
                if (state.jumpAttempts[jumpKey] or 0)>=2 then
                    releaseMovement()
                    return wait("Dua percobaan lompat belum berhasil; periksa undakan atau pindah ke jalur aman.")
                end
                local offset=edge.fromX+.5-x
                if math.abs(offset)>.15 then
                    key(Enum.KeyCode.Space,false)
                    key(Enum.KeyCode.Left,offset<0) key(Enum.KeyCode.Right,offset>0)
                    if now-state.legAt>6 then
                        releaseMovement() state.path=nil state.nextPlan=now+1
                        return wait("Pijakan lompat terhalang; menunggu jalur aman.")
                    end
                    return wait("Menempatkan karakter di tengah pijakan lompat.")
                end
                releaseMovement()
                state.jumpAttempts[jumpKey]=(state.jumpAttempts[jumpKey] or 0)+1
                state.jump={started=now,releaseAt=now+.2,crossed=false}
                key(Enum.KeyCode.Space,true)
                return wait("Melompat normal untuk keluar dari cekungan spawn.")
            end
            local jump=state.jump
            if now>=jump.releaseAt then key(Enum.KeyCode.Space,false) end
            if now-jump.started>2.5 or (now-jump.started>.45 and grounded and row>=edge.fromY) then
                releaseMovement()
                state.jump=nil state.path=nil state.nextPlan=now+1
                return wait("Lompatan belum mencapai undakan; menunggu sebelum mencoba lagi.")
            end
            if feet<=edge.y+1-.05 then jump.crossed=true end
            if jump.crossed then
                local dx=edge.x+.5-x
                key(Enum.KeyCode.Left,dx<-.15) key(Enum.KeyCode.Right,dx>.15)
                return wait("Melewati bibir undakan setelah ketinggian cukup.")
            end
            key(Enum.KeyCode.Left,false) key(Enum.KeyCode.Right,false)
            return wait("Menunggu lompatan cukup tinggi sebelum bergerak ke undakan.")
        end
        local valid=recoveryFloor(edge.x,edge.y)
        for y=edge.fromY,edge.y do
            if not recoveryBody(edge.x,y) then valid=false break end
            local support=items[fg(edge.x,y+1)]
            if y<edge.y and support and (support.solid or support.platform) then valid=false break end
        end
        local column=math.floor(x)
        if column~=edge.fromX and column~=edge.x then valid=false end
        if grounded then
            if not recoveryBody(column,row,state.startX,state.startY) then valid=false end
            local intentionalDrop=edge.y>edge.fromY and column==edge.x and row>=edge.fromY and row<edge.y
            if not intentionalDrop and not recoveryFloor(column,row) then valid=false end
        end
        if not valid or (grounded and (row<edge.fromY or row>edge.y)) then
            releaseMovement()
            state.path=nil state.nextPlan=now+1
            state.blocked="Jalur berubah; menunggu jalur kembali yang aman."
            return wait(state.blocked)
        end
        local targetX=edge.x+.5
        if grounded and row==edge.y and math.abs(x-targetX)<=.2 then
            releaseMovement()
            state.index=state.index+1
            if not state.path[state.index] then state.path=nil state.nextPlan=now end
            state.progressX,state.progressY,state.progressAt,state.legAt=x,feet,now,now
            return wait("Melanjutkan jalan ke titik farm tersimpan.")
        end
        if math.abs(x-state.progressX)>.12 or math.abs(feet-state.progressY)>.12 then
            state.progressX,state.progressY,state.progressAt=x,feet,now
        end
        if now-state.progressAt>4 or now-state.legAt>15 then
            releaseMovement()
            state.path=nil state.nextPlan=now+1
            state.blocked="Perjalanan kembali terhalang; mencoba lagi setelah jalur tersedia."
            return wait(state.blocked)
        end
        if not grounded and edge.y==edge.fromY then
            releaseMovement() state.path=nil state.nextPlan=now+1
            return wait("Menunggu mendarat; jalur kembali akan dibaca ulang.")
        end
        local dx=targetX-x
        local direction=math.abs(dx)<=.18 and 0 or (dx<0 and -1 or 1)
        -- This is the sole movement path allowed outside the farm row. It is
        -- gated by explicit recovery intent, a fresh native view and safe cells.
        if app.Recovering~=true or app.WantsRun~=true or anchorIssue() or paused() then
            adapter.Release() return wait("Perjalanan kembali dihentikan.")
        end
        key(Enum.KeyCode.Left,direction<0)
        key(Enum.KeyCode.Right,direction>0)
        adapter.TravelBusy=direction~=0
        return wait(edge.y>edge.fromY and "Turun lewat tepi menuju titik farm tersimpan." or "Berjalan kembali ke titik farm tersimpan.")
    end
    local function guard(i,overhead,build)
        if not app.Alive or not app.Engine or not app.Engine.State.running or paused() or not withinFarm() then return end
        if farmMode()=="break_only" and breakOnlyLayoutIssue(app.Anchor) then return end
        if overhead and not atAnchor() then return end
        if app.Pickup and app.Pickup.State.busy then return end
        local x,y=coords(i,overhead)
        if not x or not permitted(x,y,build) or not inRange(x,y,build) then return end
        local ready=math.max(adapter.ReadyAt,ctx.punchReady or 0)
        if os.clock() < ready then return nil,nil,ready end
        return x,y
    end
    local function recorded(kind,i)
        adapter.Recent[#adapter.Recent+1]={kind=kind,slot=i,t=os.clock()}
        if #adapter.Recent>120 then table.remove(adapter.Recent,1) end
    end
    function adapter.Plant(i)
        if farmMode()~="farm" then return false end
        local x,y,ready=guard(i,false,true)
        if not x then return false,ready end
        local tr=trees()
        if fg(x,y)~=0 or type(tr)~="table" or tr[y*ctx.w+x] or inventory:count(seedId)<1 or not canPlace(items[seedId],x,y) then return false end
        plantSeed(seedId,x,y)
        if visual and type(visual.SuppressPlantFlights)=="function" then pcall(visual.SuppressPlantFlights) end
        adapter.ReadyAt=os.clock()+.15
        local sent=fg(x,y)==seedId
        if sent then recorded("plant",i) end
        return sent,adapter.ReadyAt
    end
    function adapter.Place(i)
        local x,y,ready=guard(i,true,true)
        if not x then return false,ready end
        local tr=trees()
        if fg(x,y)~=0 or type(tr)~="table" or tr[y*ctx.w+x] or inventory:count(blockId)<1 or not canPlace(items[blockId],x,y) then return false end
        setTile("fg",x,y,blockId,false,false)
        inventory:remove(blockId,1)
        adapter.ReadyAt=os.clock()+.15
        local sent=fg(x,y)==blockId
        if sent then recorded("place",i) end
        return sent,adapter.ReadyAt
    end
    local function punch(i,overhead)
        local x,y,ready=guard(i,overhead,false)
        if not x then return false,ready end
        local id=overhead and blockId or seedId
        if fg(x,y)~=id then return false end
        local tr=trees()
        if type(tr)~="table" then return false end
        if overhead then if tr[y*ctx.w+x] then return false end
        else local t=tr[y*ctx.w+x] if not t or type(t.p)~="number" or not treeRipe(seedId,t) then return false end end
        local placed=ctx.placedAt and ctx.placedAt.fg and ctx.placedAt.fg[y*ctx.w+x]
        if placed and os.clock()<placed+.3 then return false,placed+.3 end
        local baseline,started
        if overhead then
            baseline={} started=os.clock()
            for _,d in pairs(ctx.dropList()) do if type(d)=="table" and d.uid~=nil then baseline[d.uid]=true end end
        end
        hit(x,y)
        if overhead and fg(x,y)==0 then beginBlockBreak(x,y,started,baseline) end
        ctx.punchReady=os.clock()+.22
        adapter.ReadyAt=ctx.punchReady
        local sent=overhead or fg(x,y)==0
        if sent then recorded(overhead and "punch" or "harvest",i) end
        return sent,adapter.ReadyAt
    end
    function adapter.PunchBlock(i) return punch(i,true) end
    function adapter.Harvest(i) if farmMode()~="farm" then return false end return punch(i,false) end
    function adapter.Collect(now)
        if collecting or now<nextCollect or not app.Alive or not app.Engine or not app.Engine.State.running
            or seedOverflow or paused() or not withinFarm() then return end
        refreshBlockProvenance(now)
        if next(blockClaims)==nil then return end
        nextCollect=now+.15
        local setCount=debug.getupvalues(ctx.dropTaken)[3]
        if type(setCount)~="function" or debug.info(setCount,"n")~="setDropCount"
            or debug.getupvalues(setCount)[1]~=ctx.dropList() then return end
        local collectWorld, collectJob, collectAnchor,collectEpoch = ctx.name, game.JobId, app.Anchor,provenanceEpoch
        local function currentCollection()
            return ctx.name==collectWorld and game.JobId==collectJob and app.World==collectWorld
                and app.Anchor==collectAnchor and provenanceEpoch==collectEpoch and not seedOverflow and not paused() and withinFarm()
        end
        collecting=true
        task.spawn(function()
            local ok,err=pcall(function()
                if not app.Alive or not app.Engine or not app.Engine.State.running or not currentCollection() then return end
                local candidates={}
                local px,py=position()
                for _,d in pairs(ctx.dropList()) do
                    if type(d)=="table" and pickupEligible(d,os.clock()) and not d.taking and d.n>0
                        and (d.id==112 or inventory:room(d.id)>0)
                        and (retryAt[d.uid] or 0)<=os.clock() and dropInFarmScope(d,app.Anchor)
                        and directPickupRange(d,px,py) then candidates[#candidates+1]=d end
                end
                table.sort(candidates,function(a,b) return a.uid<b.uid end)
                for index,d in ipairs(candidates) do
                    if index>5 or not app.Alive or not app.Engine or not app.Engine.State.running or not currentCollection() then break end
                    refreshBlockProvenance(os.clock())
                    local px,py=position()
                    if pickupEligible(d,os.clock()) and directPickupRange(d,px,py)
                        and not d.taking and table.find(ctx.dropList(),d) and d.n>0 then
                        local want=d.id==112 and d.n or math.min(d.n,inventory:room(d.id))
                        retryAt[d.uid]=os.clock()+1
                        if want>0 then
                            local requestedId=d.id
                            local requestedClaim=blockClaims[d.uid]
                            d.taking=true
                            adapter.LastPickupOutcome={uid=d.uid,id=requestedId,requested=want,kind="pending",at=os.clock(),world=collectWorld,job=collectJob}
                            local received,got=pcall(ctx.takeDrop,d.uid,want)
                            d.taking=false
                            local sameClaim=provenanceEpoch==collectEpoch and blockClaims[d.uid]==requestedClaim
                                and requestedClaim.object==d and requestedClaim.id==d.id and table.find(ctx.dropList(),d)
                            local outcome={uid=d.uid,id=requestedId,requested=want,at=os.clock(),world=collectWorld,job=collectJob,stale=not sameClaim}
                            if received and type(got)=="number" and got>0 then outcome.kind,outcome.amount="confirmed",got
                            elseif not received then outcome.kind,outcome.detail="error",tostring(got):sub(1,120)
                            elseif type(got)~="number" then outcome.kind,outcome.detail="non_numeric",tostring(got):sub(1,120)
                            elseif got==0 then outcome.kind,outcome.detail="zero","0"
                            elseif got<0 then outcome.kind,outcome.detail="negative",tostring(got)
                            else outcome.kind,outcome.detail="invalid_number",tostring(got) end
                            adapter.LastPickupOutcome=outcome
                            if received and type(got)=="number" and got>0 then
                                if ctx.name~=collectWorld or game.JobId~=collectJob or app.World~=collectWorld then
                                    outcome.bookkeepingSkipped=true
                                    adapter.LastCollect="Pickup world sebelumnya selesai; menunggu data tas world saat ini."
                                    break
                                end
                                -- Match native bookkeeping only after the server confirms pickup.
                                if requestedId==112 then helpers.addGems(got) else inventory:add(requestedId,got) end
                                if table.find(ctx.dropList(),d) and d.id==requestedId then setCount(d,d.n-got) end
                                adapter.Collected=adapter.Collected+got
                                local counter=requestedId==112 and "CollectedGems" or (requestedId==seedId and "CollectedSeeds" or "CollectedBlocks")
                                adapter[counter]=adapter[counter]+got
                                if sameClaim then
                                    failedPickup[d.uid]=nil
                                    retryAt[d.uid]=os.clock()+1
                                    requestedClaim.failureCount,requestedClaim.deferredUntil=0,nil
                                    requestedClaim.blockingExhausted,requestedClaim.deferReason=nil,nil
                                end
                                adapter.LastCollect="Item masuk tas: "..got
                            elseif sameClaim then
                                retryAt[d.uid]=os.clock()+2
                                requestedClaim.failureCount=(requestedClaim.failureCount or 0)+1
                                failedPickup[d.uid]=requestedClaim.failureCount
                                outcome.failures=requestedClaim.failureCount
                                adapter.LastCollect=(outcome.kind=="error" and "Permintaan pickup error: " or "Respons pickup: ")..outcome.detail.."; belum ada item dikonfirmasi."
                                if requestedClaim.failureCount>=PICKUP_FAILURE_LIMIT or requestedClaim.blockingExhausted then
                                    deferBlockPickup(d,os.clock(),"unconfirmed")
                                    adapter.LastPickupOutcome.deferredUntil=requestedClaim.deferredUntil
                                    adapter.LastCollect=adapter.LastCollect.." Coba lagi dalam 15 detik; farm lanjut."
                                end
                            end
                        end
                    end
                end
                local live={}
                for _,d in pairs(ctx.dropList()) do if type(d)=="table" then live[d.uid]=true end end
                for uid in pairs(retryAt) do if not live[uid] then retryAt[uid]=nil failedPickup[uid]=nil end end
            end)
            collecting=false
            if not ok then adapter.LastCollect="Pickup tertunda: "..tostring(err):sub(1,90) end
        end)
    end
    function adapter.CleanupBlockDrops(now)
        local function waiting(message)
            adapter.LastBlockCleanup=message
            return false,message
        end
        if not app.Alive then adapter.Release() return waiting("Farm sudah ditutup.") end
        local issue=anchorIssue()
        if issue then
            adapter.Release()
            return false,issue,true
        end
        if not app.Engine or not app.Engine.State.running then adapter.Release() return waiting("Pembersihan drop berhenti.") end
        local pause=paused()
        if pause then releaseMovement() return waiting(pause) end
        if type(now)~="number" or now~=now then now=os.clock() end
        local a=app.Anchor
        if blockCleanup and (blockCleanup.anchor~=a or blockCleanup.world~=ctx.name or blockCleanup.job~=game.JobId
            or blockCleanup.x~=a.x or blockCleanup.y~=a.y or blockCleanup.seedId~=seedId) then
            adapter.Release()
            return false,"Titik/pasangan farm berubah saat membersihkan drop; START ulang.",true
        end
        if not blockCleanup or blockCleanup.done then
            blockCleanup={anchor=a,world=ctx.name,job=game.JobId,x=a.x,y=a.y,seedId=seedId,started=now}
        end
        local cleanup=blockCleanup
        local drops,available,full,deferred=centralDrops()
        cleanup.remaining,cleanup.available,cleanup.full,cleanup.deferred=#drops,available,full,deferred
        if #drops==0 then
            releaseMovement()
            local newest
            for _,window in ipairs(breakWindows) do newest=math.max(newest or window.started,window.started) end
            if newest and now-newest<.2 then return waiting("Menunggu hasil blok yang baru dipecahkan...") end
            -- Cleanup can finish without drops yet. Keep the original bounded
            -- birth windows alive for delayed native drops; they expire in refresh.
            cleanup.done=true
            adapter.LastBlockCleanup="Drop blok tengah sudah bersih."
            return true,adapter.LastBlockCleanup
        end
        cleanup.emptySince=nil
        if available==0 then
            releaseMovement()
            cleanup.done=true
            if deferred>0 then
                -- Exhausted UIDs retry only when normally reachable. They never
                -- regain ownership of movement or block another farm batch.
                adapter.Collect(now)
                adapter.LastBlockCleanup="Pickup ditunda: "..deferred.." tumpukan tetap ada; farm lanjut."
                if full>0 then adapter.LastBlockCleanup=adapter.LastBlockCleanup.." Tas penuh: "..full.." tumpukan." end
            else adapter.LastBlockCleanup="Tas penuh: "..full.." tumpukan drop tengah ditinggal sementara; farm lanjut memakai stok." end
            return true,adapter.LastBlockCleanup
        end
        if now-cleanup.started>=CLEANUP_BLOCK_SECONDS then
            for _,d in ipairs(drops) do
                if not blockClaims[d.uid].blockingExhausted and (d.id==112 or inventory:room(d.id)>0) then
                    deferBlockPickup(d,now,"cleanup_timeout")
                end
            end
            releaseMovement()
            cleanup.done=true
            adapter.LastBlockCleanup="Batas tunggu pickup tercapai; drop tetap ada, coba lagi nanti. Farm lanjut."
            return true,adapter.LastBlockCleanup
        end
        if now-cleanup.started<.2 then releaseMovement() return waiting("Menunggu drop blok tengah muncul...") end
        local px,py=position()
        local center=a.x+.5
        if math.abs(px-center)>2.25 then releaseMovement() return waiting("Kembali ke tengah sebelum mengambil drop blok.") end
        if math.abs(py-(a.y+1))>.12 then releaseMovement() return waiting("Menunggu mendarat sebelum mengambil drop blok.") end
        if collecting then releaseMovement() return waiting("Menunggu hasil pengambilan drop blok tengah.") end
        local target,distance
        for _,d in ipairs(drops) do
            if not blockClaims[d.uid].blockingExhausted and not d.taking and not d.pending and (d.id==112 or inventory:room(d.id)>0) then
                local score=math.abs(d.x-px)+math.abs(d.y-(py-.475))
                if d.uid==cleanup.targetUid then target=d break end
                if not distance or score<distance then target,distance=d,score end
            end
        end
        if not target then releaseMovement() return waiting("Menunggu drop tengah siap diambil.") end
        if target.uid~=cleanup.targetUid then
            cleanup.targetUid=target.uid
            cleanup.routeStarted,cleanup.progressAt,cleanup.progressX=now,now,px
            cleanup.direction,cleanup.turns,cleanup.blocked=nil,0,nil
        end
        local destination=math.clamp(target.x,center-1.9,center+1.9)
        local dx=destination-px
        local direct=directPickupRange(target,px,py)
        if direct and ((failedPickup[target.uid] or 0)==0 or math.abs(dx)<=.2) then
            releaseMovement()
            adapter.Collect(now)
            return waiting("Mengambil drop blok tengah; menunggu konfirmasi masuk tas.")
        end
        -- Keep rejected/out-of-range items real. Walking can improve horizontal
        -- reach, but this routine never jumps, teleports, discards or fakes a receipt.
        if math.abs(target.y-(py-.475))>1.3 then
            releaseMovement()
            return waiting("Drop tengah belum terjangkau vertikal; menunggu turun tanpa melompat.")
        end
        if cleanup.blocked then
            local b=cleanup.blocked
            if math.abs(px-b.x)>.2 or math.abs(target.x-b.targetX)>.2 or math.abs(target.y-b.targetY)>.2 then
                cleanup.blocked=nil
                cleanup.routeStarted,cleanup.progressAt,cleanup.progressX=now,now,px
                cleanup.direction,cleanup.turns=nil,0
            else releaseMovement() return waiting("Jalur drop tengah terhalang; menunggu jalan terbuka atau posisi berubah.") end
        end
        if math.abs(dx)<=.2 then
            releaseMovement()
            return waiting("Drop tengah belum diterima; menunggu jangkauan/konfirmasi native.")
        end
        for x=math.min(math.floor(px),math.floor(destination)),math.max(math.floor(px),math.floor(destination)) do
            local support=items[fg(x,a.y+1)]
            if not permitted(x,a.y,false) or not support or (support.solid~=true and support.platform~=true)
                or support.hurts or support.deadly then
                releaseMovement()
                return waiting("Jalur drop tengah belum aman/diizinkan; farm menunggu.")
            end
        end
        if math.abs(px-cleanup.progressX)>.15 then cleanup.progressAt,cleanup.progressX=now,px end
        local direction=dx<0 and -1 or 1
        if cleanup.direction and cleanup.direction~=direction then cleanup.turns=cleanup.turns+1 end
        if now-cleanup.progressAt>4 or now-cleanup.routeStarted>10 or cleanup.turns>=3 then
            cleanup.blocked={x=px,targetX=target.x,targetY=target.y}
            releaseMovement()
            return waiting("Jalur drop tengah terhalang; menunggu jalan terbuka atau posisi berubah.")
        end
        cleanup.direction=direction
        adapter.Move(direction)
        adapter.TravelBusy=true
        return waiting("Mendekati drop blok tengah dengan jalan biasa.")
    end
    function adapter.SetCrop(id)
        if not app.Alive then return false,"UI farm sudah ditutup." end
        if app.Engine and app.Engine.State.running then return false,"STOP farm sebelum mengganti seed dan block." end
        if collecting then return false,"Tunggu pengambilan item selesai sebelum mengganti pasangan." end
        if type(id)~="number" or id%1~=0 or not cropBySeed[id] then return false,"Pasangan seed/block ini tidak didukung." end
        clearBlockProvenance()
        adapter.Release()
        selectedCrop=cropBySeed[id]
        seedId,blockId=selectedCrop.seedId,selectedCrop.blockId
        adapter.ReadyAt,adapter.Recent,adapter.Collected=0,{},0
        adapter.CollectedBlocks,adapter.CollectedSeeds,adapter.CollectedGems=0,0,0
        adapter.LastCollect=nil
        adapter.LastPickupOutcome=nil
        failedPickup,retryAt,nextCollect={},{},0
        return true,adapter.GetCrop()
    end
    return adapter
end

end)()

local makeWorldSession = (function()
-- Use the game's own world picker entry flow, including its loading flag and HUD.
return function(player, app)
    local native, menu, picker, worldMenu
    local nextDiscovery,nextModule=0,0
    local api={}
    local home,homeNative,homeVerified
    local lookupGeneration,pendingLookup,nextLookup=0,nil,0
    local homeMessage="Menunggu daftar My Worlds."
    local function recovering()
        return app.Alive==true and app.WantsRun==true and (app.Recovering==true or app.Shopping==true)
    end
    local function validWorld(value)
        return type(value)=="string" and #value>=1 and #value<=24
            and value:match("^[A-Za-z0-9]+$")~=nil
    end
    local function shopWorld()
        local value=app.ActiveVendingWorld
        if value==nil then value=app.VendingWorld end
        if value==nil then value="BLOMM" end
        if validWorld(value) and value==value:upper() then return value end
    end
    local function invalidateHome()
        lookupGeneration=lookupGeneration+1
        pendingLookup=nil homeNative=nil homeVerified=false nextLookup=0
        homeMessage="Menunggu daftar My Worlds."
    end
    local function currentPicker(value)
        local pg=player:FindFirstChildOfClass("PlayerGui")
        return value and value.Parent and pg and value:IsDescendantOf(pg)
    end
    local function loadJumpModule()
        if worldMenu or os.clock()<nextModule then return end
        nextModule=os.clock()+2
        local pg=player:FindFirstChildOfClass("PlayerGui")
        local gameGui=pg and pg:FindFirstChild("GameGui")
        local script=gameGui and gameGui:FindFirstChild("WorldMenu")
        if script then local loaded,value=pcall(require,script) if loaded then worldMenu=value end end
    end
    local function discover()
        if native and menu and currentPicker(picker) then return true end
        if native or menu or picker then
            native,menu,picker,worldMenu=nil,nil,nil,nil
            invalidateHome()
        end
        if os.clock()<nextDiscovery then return false end
        nextDiscovery=os.clock()+2
        local pg=player:FindFirstChildOfClass("PlayerGui")
        for _,v in pairs(getgc(true)) do
            if type(v)=="table" and type(rawget(v,"openPicker"))=="function"
                and type(rawget(v,"enter"))=="function" then
                local ok,u=pcall(debug.getupvalues,v.openPicker)
                local h,p=ok and u[1],ok and u[2]
                if type(h)=="table" and type(h.currentWorld)=="function" and type(h.enter)=="function"
                    and typeof(p)=="Instance" and p:IsA("GuiObject") and pg and p:IsDescendantOf(pg) then
                    native,menu,picker=h,v,p
                    invalidateHome()
                    loadJumpModule()
                    return true
                end
            end
        end
        return false
    end
    function api.View()
        if not discover() then return {ready=false,message="Menunggu menu world siap."} end
        local ok,world=pcall(native.currentWorld)
        if not ok or type(world)~="string" then return {ready=false,message="Menunggu data world."} end
        local busy=false
        local got,u=pcall(debug.getupvalues,menu.enter)
        if got then busy=u[1]==true end
        local canSpaceJump=false
        loadJumpModule()
        if worldMenu and type(worldMenu.spaceJumps)=="function" and type(native.settings)=="table" then
            local checked,value=pcall(worldMenu.spaceJumps,native.settings)
            canSpaceJump=checked and value==true
        end
        return {ready=true,world=world,picker=picker.Visible==true,loading=busy,
            typing=game:GetService("UserInputService"):GetFocusedTextBox()~=nil,canSpaceJump=canSpaceJump}
    end
    function api.RefreshMyWorld()
        -- Keep the previous owned name only as a preference. It is not trusted
        -- again until the current account's fresh native list includes it.
        invalidateHome()
    end
    function api.MyWorld()
        -- Ownership lookup is read-only and also initializes settings while OFF.
        -- Only the travel methods below require an active recovery.
        if app.Alive~=true then
            if pendingLookup then invalidateHome() end
            return nil,"Sesi telah ditutup."
        end
        if not discover() then return nil,"Menunggu menu world siap." end
        if homeVerified and homeNative==native then return home end
        if pendingLookup and os.clock()-pendingLookup.started>=15 then
            invalidateHome()
            nextLookup=os.clock()+5
            homeMessage="My Worlds timeout; mencoba lagi."
            return nil,homeMessage
        end
        if pendingLookup or os.clock()<nextLookup then return nil,homeMessage end
        if type(native.list)~="function" then
            nextLookup=os.clock()+5
            homeMessage="Daftar My Worlds belum tersedia."
            return nil,homeMessage
        end
        local token={generation=lookupGeneration,native=native,menu=menu,picker=picker,list=native.list,
            started=os.clock()}
        pendingLookup=token
        homeMessage="Memuat My Worlds akun ini..."
        -- ListWorlds can yield on the network. Never perform that lookup in the
        -- caller's heartbeat or let an older response replace a refreshed list.
        task.spawn(function()
            if pendingLookup~=token or lookupGeneration~=token.generation then return end
            if app.Alive~=true or native~=token.native or menu~=token.menu
                or picker~=token.picker or native.list~=token.list or not currentPicker(token.picker) then
                pendingLookup=nil
                return
            end
            local ok,result=pcall(token.list)
            if pendingLookup~=token or lookupGeneration~=token.generation then return end
            pendingLookup=nil
            if app.Alive~=true or native~=token.native or menu~=token.menu
                or picker~=token.picker or native.list~=token.list or not currentPicker(token.picker) then
                homeVerified=false homeNative=nil
                homeMessage="Menunggu daftar My Worlds."
                return
            end
            nextLookup=os.clock()+5
            if not ok or type(result)~="table" or type(result.mine)~="table" then
                homeMessage="Daftar My Worlds belum tersedia; mencoba lagi."
                return
            end
            local first,preferred
            for _,name in ipairs(result.mine) do
                if validWorld(name) then
                    first=first or name
                    if name==home then preferred=name end
                end
            end
            local chosen=preferred or first
            if not chosen then
                homeMessage="Belum ada world milik akun ini di My Worlds."
                return
            end
            home=chosen homeNative=native homeVerified=true
            homeMessage=nil
        end)
        if homeVerified and homeNative==native then return home end
        return nil,homeMessage
    end
    function api.OpenPicker()
        local view=api.View()
        if not recovering() then return false,"Pemulihan dibatalkan." end
        if not view.ready or view.loading or view.typing then return false,"Menunggu menu world siap." end
        if view.picker then return true,"Menu world sudah terbuka." end
        local ok=pcall(menu.openPicker)
        if not ok then return false,"Menu world belum bisa dibuka." end
        return true
    end
    function api.Enter(world)
        local view=api.View()
        if not recovering() then return false,"Pemulihan dibatalkan." end
        if not validWorld(world) then return false,"Nama world tidak valid." end
        local shoppingDestination=app.Shopping==true and world==shopWorld()
        if not shoppingDestination and (not homeVerified or homeNative~=native or world~=home) then
            return false,"World tujuan belum terverifikasi di My Worlds akun ini."
        end
        if not view.ready or not view.picker or view.loading or view.typing then return false,"Menunggu menu world siap." end
        -- The observed native WorldMenu.enter(worldName) calls handlers.enter,
        -- closes the picker only on success, and restores the game's own HUD.
        local ok=pcall(menu.enter,world)
        if not ok then return false,"World belum bisa dimasuki." end
        return true
    end
    return api
end

end)()

local makeConfigStore = (function()
-- Account/place-local preferences and resume intent. No game actions or runtime state.
return function(options)
    options=type(options)=="table" and options or {}
    local MAX_INTEGER=9007199254740991
    local function integer(value,low,high)
        return type(value)=="number" and value==value and value%1==0 and value>=low and value<=high
    end
    local function mode(value) return value=="farm" or value=="break_only" end
    local function world(value)
        return type(value)=="string" and #value>=1 and #value<=24 and value:match("^[A-Z0-9]+$")~=nil
    end
    local identityValid=integer(options.userId,1,MAX_INTEGER) and integer(options.placeId,1,MAX_INTEGER)
    local prefix=identityValid and ("BLUE_HUB_FARM_CONFIG_"..string.format("%.0f",options.userId)
        .."_"..string.format("%.0f",options.placeId)) or nil
    local store={Path=prefix and prefix.."_a.json" or nil,BackupPath=prefix and prefix.."_b.json" or nil,
        LegacyPath="BLUE_HUB_FARM_SELECTION.json"}
    local paths={store.Path,store.BackupPath}
    local function preferences(value)
        local result={}
        if type(value)~="table" then return result end
        if integer(value.seedId,2,2147483647) then result.seedId=value.seedId end
        if mode(value.mode) then result.mode=value.mode end
        if integer(value.plantCount,1,99) then result.plantCount=value.plantCount end
        return result
    end
    local function rowKey(plan)
        local a=plan.anchor
        local first,last,y=a.x-1,a.x+1,a.y-1
        local head=plan.world..":blocks:3:"
        if plan.mode=="farm" then
            local left=math.floor((plan.plantCount-1)/2)
            first,last,y=a.x-left,a.x+plan.plantCount-1-left,a.y
            head=plan.world..":"..plan.plantCount..":"
        end
        return head..first..":"..y..":"..last..":"..y..":"..plan.seedId..":"..(plan.seedId-1)
    end
    local function progress(value,plan)
        if value==nil then return nil,true end
        if type(value)~="table" or value.version~=1 or value.world~=plan.world or value.mode~=plan.mode
            or value.rowKey~=rowKey(plan) or value.blockCount~=3
            or value.seedCount~=(plan.mode=="farm" and plan.plantCount or 0) then return nil,false end
        local result={version=1,world=plan.world,mode=plan.mode,rowKey=rowKey(plan),
            blockCount=3,seedCount=plan.mode=="farm" and plan.plantCount or 0,phase=value.phase}
        if plan.mode=="break_only" then
            if value.phase~="blocks" then return nil,false end
        else
            if value.phase~="planting" and value.phase~="growing" and value.phase~="harvesting" then return nil,false end
            if value.rowDirection~=nil and value.rowDirection~=1 and value.rowDirection~=-1 then return nil,false end
            if value.rowCursor~=nil and not integer(value.rowCursor,0,plan.plantCount+1) then return nil,false end
            result.rowDirection=value.rowDirection or 1
            result.rowCursor=math.max(1,math.min(plan.plantCount,value.rowCursor or (result.rowDirection==1 and 1 or plan.plantCount)))
        end
        if value.plantingStarted~=nil and type(value.plantingStarted)~="boolean" then return nil,false end
        if value.blockCursor~=nil and not integer(value.blockCursor,1,3) then return nil,false end
        result.plantingStarted=value.plantingStarted==true and value.phase=="planting"
        result.blockCursor=value.blockCursor or 1
        return result,true
    end
    local function plan(value,prefs)
        if type(value)~="table" or not world(value.world) or value.seedId~=prefs.seedId
            or not mode(value.mode) or value.mode~=prefs.mode or value.plantCount~=prefs.plantCount
            or not integer(value.plantCount,1,99) or not integer(value.seedId,2,2147483647) then return nil end
        local a=value.anchor
        if type(a)~="table" or a.world~=value.world or a.mode~=value.mode or a.plantCount~=value.plantCount
            or not integer(a.x,1,4094) or not integer(a.y,1,4094)
            or (a.seedId~=nil and a.seedId~=value.seedId) or (a.blockId~=nil and a.blockId~=value.seedId-1) then return nil end
        local left=math.floor((value.plantCount-1)/2)
        if value.mode=="farm" and (a.x-left<0 or a.x+value.plantCount-1-left>4095) then return nil end
        local result={world=value.world,seedId=value.seedId,mode=value.mode,plantCount=value.plantCount,
            anchor={world=value.world,x=a.x,y=a.y,mode=value.mode,plantCount=value.plantCount,
                seedId=value.seedId,blockId=value.seedId-1}}
        local hint,valid=progress(value.progress,result)
        if not valid then return nil end
        result.progress=hint
        return result
    end
    local function settings(value)
        local result=preferences(value)
        result.autoResume=false
        if type(value)~="table" then return result,true end
        local validPrefs=result.seedId~=nil and result.mode~=nil and result.plantCount~=nil
        local run=validPrefs and plan(value.lastRun,result) or nil
        result.lastRun=run
        local invalid=value.lastRun~=nil and not run or value.autoResume~=nil and type(value.autoResume)~="boolean"
        if value.autoResume==true and not run then invalid=true end
        if validPrefs and not invalid then result.autoResume=value.autoResume==true end
        return result,invalid or not validPrefs
    end
    local function record(value)
        if type(value)~="table" or value.schema~="BLUE_HUB_FARM_CONFIG" or value.userId~=options.userId
            or value.placeId~=options.placeId or not integer(value.revision,1,MAX_INTEGER-1) then return nil end
        local clean,invalid=settings(value.settings)
        if value.version~=1 then clean=preferences(value.settings);clean.autoResume=false;invalid=true end
        return {revision=value.revision,settings=clean,invalid=invalid,version=value.version}
    end
    local function readRecord(path)
        local ok,raw=pcall(options.read,path)
        if not ok then
            local detail=tostring(raw):lower()
            local absent=detail:find("no such file",1,true) or detail:find("does not exist",1,true)
                or detail:find("file not found",1,true) or detail:find("cannot find the file",1,true)
            return nil,absent and "missing" or "unverifiable"
        end
        if type(raw)~="string" or raw=="" then return nil,"invalid" end
        local decoded,value=pcall(options.decode,raw)
        if not decoded then return nil,"invalid" end
        local candidate=record(value)
        return candidate,candidate and "valid" or "invalid"
    end
    local function same(a,b)
        if type(a)~=type(b) then return false end
        if type(a)~="table" then return a==b end
        for key,value in pairs(a) do if not same(value,b[key]) then return false end end
        for key in pairs(b) do if a[key]==nil then return false end end
        return true
    end
    local function readBoth()
        local records,states={},{}
        for i,path in ipairs(paths) do records[i],states[i]=readRecord(path) end
        local best,index
        for i=1,2 do
            local candidate=records[i]
            if candidate and (not best or candidate.revision>best.revision
                or candidate.revision==best.revision and best.settings.autoResume and not candidate.settings.autoResume) then
                best,index=candidate,i
            end
        end
        if best and records[1] and records[2] and records[1].revision==records[2].revision
            and not same(records[1].settings,records[2].settings) then
            -- A conflicting same-revision pair cannot establish running intent.
            best.settings.autoResume=false
            best.invalid=true
        end
        return records,best,index,states
    end
    local function readable()
        if not identityValid then return false,"Identitas akun/place tidak valid; config tidak diakses." end
        if type(options.read)~="function" or type(options.decode)~="function" then
            return false,"Penyimpanan tidak tersedia: perlu readfile dan JSON decode."
        end
        return true
    end
    function store.Load()
        local canRead,message=readable()
        if not canRead then return {},message end
        local records,best,_,states=readBoth()
        if best then
            local status="Config dimuat (revisi "..best.revision..")."
            if not records[1] or not records[2] then status="Config dipulihkan dari salinan valid (revisi "..best.revision..")." end
            if best.invalid then status=status.." Data tidak valid/tidak dikenal; auto resume OFF." end
            if states[1]=="invalid" or states[1]=="unverifiable" or states[2]=="invalid" or states[2]=="unverifiable" then
                -- A corrupt slot may be an interrupted newer STOP. Recover data,
                -- but never resurrect RUN from the remaining older record.
                best.settings.autoResume=false
                status=status.." Salinan lain rusak/tidak dapat diverifikasi; auto resume OFF."
            end
            return best.settings,status
        end
        local ok,raw=pcall(options.read,store.LegacyPath)
        if ok and type(raw)=="string" then
            local decoded,value=pcall(options.decode,raw)
            if decoded and type(value)=="table" then
                local prefs=preferences(value)
                if next(prefs) then
                    prefs.autoResume=false
                    return prefs,"Pilihan lama dimuat; titik dan auto resume tidak dimigrasikan."
                end
            end
        end
        return {},"Belum ada config valid untuk akun/place ini."
    end
    function store.Save(value)
        local canRead,message=readable()
        if not canRead then return false,message end
        if type(options.write)~="function" or type(options.encode)~="function" then
            return false,"Penyimpanan tidak tersedia: perlu writefile dan JSON encode."
        end
        local clean,invalid=settings(value)
        if not clean.seedId or not clean.mode or not clean.plantCount then
            return false,"Pilihan seed, mode, atau jumlah tanaman tidak valid; config belum disimpan."
        end
        local records,best=readBoth()
        local revision=best and best.revision+1 or 1
        if not integer(revision,1,MAX_INTEGER-1) then return false,"Revisi config tidak valid; belum disimpan." end
        local payload={schema="BLUE_HUB_FARM_CONFIG",version=1,userId=options.userId,placeId=options.placeId,
            revision=revision,settings=clean}
        local encoded,raw=pcall(options.encode,payload)
        if not encoded or type(raw)~="string" or raw=="" then return false,"JSON config gagal dibuat; belum disimpan." end
        local decoded,check=pcall(options.decode,raw)
        local checked=decoded and record(check) or nil
        if not checked or checked.revision~=revision or not same(checked.settings,clean) then
            return false,"JSON config tidak lolos validasi; belum disimpan."
        end
        -- Write the older/absent slot first; never destroy the sole valid copy
        -- before trying to establish its replacement. Every successful slot is read back.
        local first=1
        if records[1] and (not records[2] or records[1].revision>records[2].revision) then first=2 end
        local confirmed,replacementEstablished=0,false
        for order,i in ipairs({first,3-first}) do
            if order==2 and best and records[i] and not replacementEstablished then break end
            local wrote=pcall(options.write,paths[i],raw)
            local readBack,bytes=pcall(options.read,paths[i])
            local saved=readBack and bytes==raw and readRecord(paths[i]) or nil
            if saved and saved.revision==revision and same(saved.settings,clean) then
                replacementEstablished=true
                if wrote then confirmed=confirmed+1 end
            end
        end
        if confirmed==0 then return false,"Config gagal ditulis/diverifikasi; belum tersimpan." end
        local status=confirmed==2 and "Config dan salinan cadangan tersimpan." or "Config tersimpan satu salinan; cadangan gagal diverifikasi."
        if invalid then status=status.." Rencana resume tidak valid; disimpan dengan auto resume OFF." end
        return true,status
    end
    return store
end

end)()

local makeCommerceAdapter = (function()
-- Native shop/vending boundary. Reads never block the heartbeat on a network RPC.
return function(player,app)
    local api={Context=nil,InFlight=false}
    local closed=false
    local native,menu,picker,inventory,store,itemDB
    local nextDiscovery=0
    local wallet,request,nextWallet,generation=nil,nil,0,0
    local walletMessage="Memuat saldo gems..."
    local cachedVendor,nextVendor=nil,0
    local WL,GRASS=242,16
    local function shopWorld()
        -- The coordinator can retain the old journal's world while reconciling
        -- an already-sent transaction after the loader config changes.
        local value=app.ActiveVendingWorld
        if value==nil then value=app.VendingWorld end
        if value==nil then value="BLOMM" end
        if type(value)=="string" and #value>0 and #value<=24 and value:match("^[A-Z0-9]+$") then return value end
    end
    local function finite(n) return type(n)=="number" and n==n and n>-math.huge and n<math.huge end
    local function integer(n) return finite(n) and n>=0 and n==math.floor(n) end
    local function authorized() return not closed and app.Alive==true and app.WantsRun==true and app.Shopping==true and shopWorld()~=nil end
    local function invalidateWallet()
        generation=generation+1 wallet=nil request=nil nextWallet=0
        walletMessage="Memuat saldo gems..."
    end
    local function currentPicker(p)
        local pg=player:FindFirstChildOfClass("PlayerGui")
        return p and p.Parent and pg and p:IsDescendantOf(pg)
    end
    local function discover()
        if closed then return false end
        if native and menu and currentPicker(picker) and inventory and api.Context then return true end
        if native or menu or picker then
            native,menu,picker,inventory,store,itemDB=nil,nil,nil,nil,nil,nil
            api.Context=nil cachedVendor=nil nextVendor=0 invalidateWallet()
        end
        if os.clock()<nextDiscovery then return false end
        nextDiscovery=os.clock()+2
        local pg=player:FindFirstChildOfClass("PlayerGui")
        if not pg then return false end
        for _,candidate in pairs(getgc(true)) do
            if type(candidate)=="table" and type(rawget(candidate,"openPicker"))=="function"
                and type(rawget(candidate,"enter"))=="function" then
                local ok,u=pcall(debug.getupvalues,candidate.openPicker)
                local h,p=ok and u[1],ok and u[2]
                if type(h)=="table" and type(h.buy)=="function" and type(h.gems)=="function"
                    and type(h.enter)=="function" and type(h.currentWorld)=="function"
                    and typeof(p)=="Instance" and p:IsA("GuiObject") and p:IsDescendantOf(pg) then
                    local got,buy=pcall(debug.getupvalues,h.buy)
                    local entered,e=pcall(debug.getupvalues,h.enter)
                    local inv,ctx=got and buy[1],got and buy[2]
                    if entered and type(ctx)=="table" and e[1]==ctx and type(inv)=="table"
                        and type(inv.count)=="function" and type(inv.room)=="function" then
                        local rs=game:GetService("ReplicatedStorage")
                        local loaded,s,db=pcall(function()
                            return require(rs.Shared.Store),require(rs.Shared.ItemDB)
                        end)
                        if loaded and type(s)=="table" and type(db)=="table" then
                            native,menu,picker,inventory,store,itemDB=h,candidate,p,inv,s,db
                            api.Context=ctx invalidateWallet()
                            return true
                        end
                    end
                end
            end
        end
        return false
    end
    local function liveWorld()
        if not discover() then return nil,"Menunggu shop game siap." end
        local ok,world=pcall(native.currentWorld)
        local got,u=pcall(debug.getupvalues,menu.enter)
        if not ok or type(world)~="string" or world=="" or world~=api.Context.name
            or picker.Visible or not got or u[1]==true then
            return nil,"Menunggu world selesai dimuat."
        end
        return world
    end
    local function price()
        local offer=store.byKey and store.byKey.world_lock
        local active=false
        for _,v in ipairs(store.OFFERS or {}) do if v==offer then active=true break end end
        if not active or type(offer)~="table" or offer.key~="world_lock" or offer.robux or offer.product
            or type(offer.items)~="table" or #offer.items~=1 or type(offer.items[1])~="table"
            or offer.items[1][1]~=WL or offer.items[1][2]~=1 then
            return nil,"Offer World Lock belum terverifikasi."
        end
        local raw=game:GetService("ReplicatedStorage"):GetAttribute("StorePrices")
        local overrides={}
        if raw~=nil then
            if type(raw)~="string" then return nil,"Data harga shop belum valid." end
            local http=game:GetService("HttpService")
            local ok,decoded=pcall(http.JSONDecode,http,raw)
            if not ok or type(decoded)~="table" then return nil,"Data harga shop belum valid." end
            overrides=decoded
        end
        local quoted=tonumber(overrides.world_lock)
        if quoted~=nil and (not finite(quoted) or quoted<0) then return nil,"Harga World Lock belum valid." end
        -- Mirrors StorePriceSync, including the no-override base-price fallback.
        local value=quoted~=nil and math.floor(quoted) or (offer.basePrice or offer.price)
        if not integer(value) or value<=0 then return nil,"Harga World Lock belum valid." end
        return value
    end
    local function inventoryRead()
        local ok,wl,grass,wlRoom,grassRoom=pcall(function()
            return inventory:count(WL),inventory:count(GRASS),inventory:room(WL),inventory:room(GRASS)
        end)
        if not ok or not integer(wl) or not integer(grass) or not integer(wlRoom) or not integer(grassRoom) then
            return nil,"Data tas belum valid."
        end
        return {wl=wl,grass=grass,wlRoom=wlRoom,grassRoom=grassRoom}
    end
    local function requestWallet(world)
        if request and os.clock()-request.started>=15 then
            invalidateWallet() nextWallet=os.clock()+5
            walletMessage="Saldo gems timeout; mencoba lagi."
        end
        if request or os.clock()<nextWallet then return end
        local token={generation=generation,native=native,menu=menu,picker=picker,
            context=api.Context,world=world,job=game.JobId,fn=native.gems,started=os.clock()}
        request=token
        task.spawn(function()
            if request~=token or generation~=token.generation then return end
            if closed or app.Alive~=true or not currentPicker(token.picker) then request=nil return end
            local ok,gems=pcall(token.fn)
            if request~=token or generation~=token.generation then return end
            request=nil
            if closed or app.Alive~=true or native~=token.native or menu~=token.menu or picker~=token.picker
                or api.Context~=token.context or not currentPicker(token.picker)
                or game.JobId~=token.job or api.Context.name~=token.world or native.gems~=token.fn then return end
            if not ok or not finite(gems) or gems<0 then
                nextWallet=os.clock()+5 walletMessage="Saldo gems belum tersedia; mencoba lagi."
                return
            end
            wallet={gems=math.floor(gems),at=os.clock(),world=world,job=game.JobId,native=native}
            walletMessage=nil
        end)
    end
    function api.Read()
        if closed or app.Alive~=true then return {ready=false,message="Sesi ditutup."} end
        local world,message=liveWorld()
        if not world then return {ready=false,message=message} end
        local cost,why=price()
        if not cost then return {ready=false,world=world,message=why} end
        local counts,err=inventoryRead()
        if not counts then return {ready=false,world=world,message=err} end
        counts.world=world counts.price=cost
        if api.InFlight then counts.ready=false counts.message="Menunggu transaksi sebelumnya." return counts end
        if not wallet or wallet.native~=native or wallet.world~=world or wallet.job~=game.JobId
            or os.clock()-wallet.at>=5 then
            wallet=nil requestWallet(world)
        end
        if not wallet then counts.ready=false counts.message=walletMessage return counts end
        counts.ready=true counts.gems=wallet.gems
        return counts
    end
    function api.RefreshWallet()
        invalidateWallet()
    end
    function api.ReadItem(id)
        if closed or app.Alive~=true then return nil,"Sesi ditutup." end
        local world,message=liveWorld()
        if not world then return nil,message end
        if not integer(id) or id<1 or type(itemDB[id])~="table" then
            return nil,"Item vending tidak dikenal pada katalog game."
        end
        local ctx,inv,job=api.Context,inventory,game.JobId
        local ok,count,room=pcall(function() return inv:count(id),inv:room(id) end)
        if not ok or not integer(count) or not integer(room) then return nil,"Data tas item belum valid." end
        if closed or app.Alive~=true or api.Context~=ctx or inventory~=inv or game.JobId~=job or ctx.name~=world then
            return nil,"Sesi berubah saat membaca tas item."
        end
        return {itemId=id,count=count,room=room,name=tostring(itemDB[id].name or id)}
    end
    function api.Position()
        local world,message=liveWorld()
        if not world then return nil,nil,message end
        if type(api.Context.atTile)~="function" then return nil,nil,"Posisi karakter belum tersedia." end
        local ok,u=pcall(debug.getupvalues,api.Context.atTile)
        if not ok or not finite(u[1]) or not finite(u[2]) then return nil,nil,"Posisi karakter belum tersedia." end
        return u[1],u[2]
    end
    local function recordAt(ctx,x,y)
        local job,target=game.JobId,shopWorld()
        if not target or ctx.name~=target then return nil,"World vending tidak sesuai config." end
        if not integer(x) or not integer(y) or not integer(ctx.w) or not integer(ctx.h) or x>=ctx.w or y>=ctx.h
            or type(ctx.fg)~="table" or type(ctx.fg[y])~="table"
            or type(ctx.vends)~="table" or type(ctx.vends.get)~="function" then return nil,"Data vending belum tersedia." end
        local tile=itemDB[ctx.fg[y][x]]
        if not tile or not tile.vend then return nil,"Vending sudah berubah." end
        local ok,r=pcall(ctx.vends.get,x,y)
        if closed or api.Context~=ctx or ctx.name~=target or shopWorld()~=target or game.JobId~=job then
            return nil,"Sesi vending berubah."
        end
        if not ok or type(r)~="table" or r.shop or not integer(r.id) or r.id<1 or type(itemDB[r.id])~="table" or not integer(r.stock)
            or r.stock<=0 or r.cost~=1 or r.per~=1 then
            return nil,"Item dengan harga 1 WL per item tidak tersedia di vending ini."
        end
        return r
    end
    local function vendorRecord(v)
        local world,message=liveWorld()
        if not world then return nil,message end
        if world~=shopWorld() or type(v)~="table" or v.world~=world or v.job~=game.JobId
            or v._context~=api.Context or not integer(v.x) or not integer(v.y)
            or not integer(v.itemId) or v.itemId<1 or v.id~=v.itemId or v.cost~=1 or v.per~=1 then
            return nil,"Tujuan vending tidak valid."
        end
        local r,why=recordAt(api.Context,v.x,v.y)
        if not r then return nil,why end
        if v.key~=string.format("%s:%d:%d",world,v.x,v.y) or r.id~=v.itemId then
            return nil,"Isi vending berubah; baca penawaran terbaru dahulu."
        end
        return r
    end
    local function quoteAt(ctx,x,y)
        local r,message=recordAt(ctx,x,y)
        if not r then return nil,message end
        return {world=ctx.name,job=game.JobId,x=x,y=y,id=r.id,itemId=r.id,
            name=tostring(itemDB[r.id].name or r.id),stock=r.stock,cost=r.cost,per=r.per,
            key=string.format("%s:%d:%d",ctx.name,x,y),_context=ctx}
    end
    function api.FindVendor(targetKey)
        local world,message=liveWorld()
        local target=shopWorld()
        if not target then return nil,"World vending pada config tidak valid." end
        if world~=target then return nil,message or ("Vending dicari hanya di "..target..".") end
        local ctx=api.Context
        if targetKey~=nil then
            if type(targetKey)~="string" or #targetKey>64 then return nil,"Identitas vending tidak valid." end
            local keyWorld,sx,sy=targetKey:match("^([A-Z0-9]+):(%d+):(%d+)$")
            local x,y=tonumber(sx),tonumber(sy)
            if keyWorld~=target or not integer(x) or not integer(y) or not integer(ctx.w) or not integer(ctx.h)
                or x>=ctx.w or y>=ctx.h or targetKey~=string.format("%s:%d:%d",target,x,y) then
                return nil,"Identitas vending tidak valid."
            end
            -- A saved machine never silently redirects to another owner's vendor.
            cachedVendor={x=x,y=y,world=target,job=game.JobId,_context=ctx}
            local v,why=quoteAt(ctx,x,y)
            return v,why
        end
        if cachedVendor and cachedVendor.world==target and cachedVendor._context==ctx and cachedVendor.job==game.JobId then
            -- Return a fresh quote object: the previous item/price remains pinned
            -- for an in-progress purchase even when this machine is restocked.
            return quoteAt(ctx,cachedVendor.x,cachedVendor.y)
        end
        if os.clock()<nextVendor then return nil,"Menunggu vending 1 WL per item." end
        nextVendor=os.clock()+1
        if type(ctx.fg)~="table" then return nil,"Data world belum tersedia." end
        if type(ctx.vends)~="table" or type(ctx.vends.get)~="function" then return nil,"Data vending belum tersedia." end
        local found,job=nil,game.JobId
        for y,row in pairs(ctx.fg) do
            if integer(y) and type(row)=="table" then
                for x,tileId in pairs(row) do
                    local tile=itemDB[tileId]
                    if integer(x) and tile and tile.vend then
                        local ok,r=pcall(ctx.vends.get,x,y)
                        if not ok or ctx~=api.Context or ctx.name~=target or shopWorld()~=target or game.JobId~=job then
                            return nil,"Data vending belum tersedia."
                        end
                        if not (type(r)=="table" and r.shop==true) then
                            if found then return nil,"Ada lebih dari satu vending di "..target.."; tujuan belum pasti." end
                            found={x=x,y=y,world=target,job=job,_context=ctx}
                        end
                    end
                end
            end
        end
        if found then
            cachedVendor=found
            return quoteAt(ctx,found.x,found.y)
        end
        return nil,"Belum ada vending dengan harga 1 WL per item."
    end
    function api.NearVendor(v)
        local r,message=vendorRecord(v)
        if not r then return false,message end
        if type(api.Context.atTile)~="function" then return false,"Posisi vending belum tersedia." end
        local ok,near=pcall(api.Context.atTile,v.x,v.y)
        return ok and near==true,"Dekati vending terlebih dahulu."
    end
    local function transaction(kind,fn,...)
        if api.InFlight then return false,"Transaksi lain masih menunggu." end
        api.InFlight=true
        api.LastOutcome={kind="pending",action=kind,world=api.Context.name,job=game.JobId}
        local ok,accepted,a,b=pcall(fn,...)
        api.InFlight=false invalidateWallet()
        api.LastOutcome.kind=not ok and "error" or (accepted==true and "confirmed" or "rejected")
        -- Native callbacks already apply their accepted inventory/gem receipts,
        -- including a receipt which returns after the user presses STOP.
        if not ok then return false,"Hasil transaksi belum pasti; periksa saldo sebelum mengulang.",nil,true end
        -- vendCall collapses a lost reply and a server refusal to the same
        -- false return. Preserve that ambiguity rather than replay a purchase.
        local uncertain=accepted~=true and (kind=="item" or kind=="grass" or a==nil
            or a=="Couldn't reach the server, try again.")
        return accepted,a,b,uncertain or nil
    end
    function api.BuyWL()
        if not authorized() then return false,"Belanja dibatalkan." end
        local view=api.Read()
        if not view.ready then return false,view.message end
        if view.wlRoom<1 then return false,"Tas World Lock penuh." end
        if view.gems<view.price then return false,"Gems belum cukup." end
        if not authorized() or api.Context.name~=view.world then return false,"World berubah sebelum pembelian." end
        return transaction("world_lock",native.buy,"world_lock")
    end
    function api.BuyItem(v,quantity)
        if not authorized() then return false,"Belanja dibatalkan." end
        if not integer(quantity) or quantity<1 then return false,"Jumlah item tidak valid." end
        local r,message=vendorRecord(v)
        if not r then return false,message end
        local near,why=api.NearVendor(v)
        if not near then return false,why end
        local item,err=api.ReadItem(v.itemId)
        if not item then return false,err end
        local ctx,inv,job=api.Context,inventory,game.JobId
        local counted,wl=pcall(inv.count,inv,WL)
        if not counted or not integer(wl) then return false,"Saldo World Lock belum valid." end
        -- Spend actual WL only. Never infer an affordable budget from DL/BGL.
        if quantity>r.stock or quantity>item.room or quantity>wl then
            return false,"Jumlah melebihi stok, ruang tas, atau World Lock tersedia."
        end
        local fn=ctx.vendCall
        if type(fn)~="function" then return false,"Vending belum siap." end
        local ok,up=pcall(debug.getupvalues,fn)
        if not ok or up[3]~=inv then return false,"Tas vending tidak cocok dengan sesi ini." end
        -- Inventory access may yield. Check the current item, quote, proximity,
        -- native binding and intent again immediately before the native call.
        local latest,lastWhy=vendorRecord(v)
        if not latest or quantity>latest.stock then return false,lastWhy or "Stok vending berubah." end
        local stillNear=api.NearVendor(v)
        if not stillNear then return false,"Dekati vending terlebih dahulu." end
        local finalItem,finalWhy=api.ReadItem(v.itemId)
        if not finalItem or quantity>finalItem.room then return false,finalWhy or "Ruang tas berubah." end
        local finalOK,finalWl=pcall(inv.count,inv,WL)
        if not finalOK or not integer(finalWl) or quantity>finalWl then return false,"Saldo World Lock berubah." end
        latest,lastWhy=vendorRecord(v)
        if not latest or quantity>latest.stock then return false,lastWhy or "Stok vending berubah." end
        if not api.NearVendor(v) then return false,"Dekati vending terlebih dahulu." end
        local rebound,finalUp=pcall(debug.getupvalues,fn)
        if not rebound or finalUp[3]~=inv then return false,"Tas vending berubah sebelum pembelian." end
        if not authorized() or api.Context~=ctx or inventory~=inv or ctx.name~=v.world or shopWorld()~=v.world
            or game.JobId~=job or ctx.vendCall~=fn then return false,"Sesi berubah sebelum pembelian." end
        return transaction("item",fn,"buy",v.x,v.y,quantity,{cost=latest.cost,per=latest.per})
    end
    function api.BuyGrass(v,quantity)
        if type(v)~="table" or v.itemId~=GRASS then return false,"Penawaran bukan Grass." end
        return api.BuyItem(v,quantity)
    end
    function api.Close()
        closed=true invalidateWallet() cachedVendor=nil api.Context=nil
    end
    return api
end

end)()

local makeCommerceTravel = (function()
-- Customer walking only. Never changes map tiles, native coordinates or farm anchor.
return function(player, app, commerce)
    local api = {}
    local items = require(game:GetService("ReplicatedStorage").Shared.ItemDB)
    local input = game:GetService("UserInputService")
    local virtualInput = game:GetService("VirtualInputManager")
    local getIdentity = getthreadidentity or getidentity
    local setIdentity = setthreadidentity or setidentity
    local held, state = {}, nil
    local function finite(n) return type(n)=="number" and n==n and math.abs(n)<math.huge end
    local function shopWorld()
        local value=app.ActiveVendingWorld
        if value==nil then value=app.VendingWorld end
        if value==nil then value="BLOMM" end
        if type(value)=="string" and #value>0 and #value<=24 and value:match("^[A-Z0-9]+$") then return value end
    end
    local function active()
        return app.Alive==true and app.WantsRun==true and app.Shopping==true
    end
    local function key(code, down)
        if held[code]==down or (not down and not held[code]) then return end
        local previous
        if type(getIdentity)=="function" then local ok,v=pcall(getIdentity) if ok then previous=v end end
        if previous~=nil and type(setIdentity)=="function" then pcall(setIdentity,8) end
        local ok,err=pcall(function() virtualInput:SendKeyEvent(down,code,false,game) end)
        if previous~=nil and type(setIdentity)=="function" then pcall(setIdentity,previous) end
        if not ok then error(err) end
        held[code]=down or nil
    end
    local function releaseKeys()
        for _,code in ipairs({Enum.KeyCode.Left,Enum.KeyCode.Right,Enum.KeyCode.Space}) do pcall(key,code,false) end
    end
    function api.Release() releaseKeys() state=nil end
    function api.Pause()
        releaseKeys()
        -- Wallet reads can briefly pause movement. Keep the trip deadlines so
        -- recurring cache refreshes cannot reset an unreachable-route timeout.
        if state then state.path=nil state.jump=nil state.nextPlan=os.clock() end
    end
    local function wait(message,fatal)
        if fatal then api.Release() end
        return false,message,fatal==true
    end
    local function step(vendor,now)
        if not active() then api.Release() return wait("Perjalanan belanja dihentikan.") end
        if not finite(now) then now=os.clock() end
        local ctx=commerce.Context
        local target=shopWorld()
        if not target or type(vendor)~="table" or not finite(vendor.x) or not finite(vendor.y)
            or vendor.x%1~=0 or vendor.y%1~=0 or vendor.world~=target then
            return wait("Tujuan vending tidak valid.",true)
        end
        if type(ctx)~="table" or ctx.name~=target or vendor.job~=game.JobId then
            return wait("World berubah saat menuju vending.",true)
        end
        local view=app.Session and app.Session.View and app.Session.View()
        if type(view)~="table" or not view.ready or view.world~=target or view.loading or view.picker then
            releaseKeys() return wait("Menunggu world "..target.." siap.")
        end
        if not active() then api.Release() return wait("Perjalanan belanja dihentikan.") end
        if state and (state.ctx~=ctx or state.job~=game.JobId or state.x~=vendor.x or state.y~=vendor.y) then
            return wait("Tujuan perjalanan vending berubah.",true)
        end
        if not state then state={ctx=ctx,job=game.JobId,x=vendor.x,y=vendor.y,started=now,nextPlan=now,lastNow=now} end
        local s=state
        local elapsed=math.max(0,now-s.lastNow)
        s.lastNow=now
        if view.typing or input:GetFocusedTextBox() or player:GetAttribute("Frozen")==true then
            releaseKeys()
            -- A deliberate typing pause should not exhaust the trip timeout.
            for _,field in ipairs({"started","blockedAt","progressAt","legAt"}) do
                if s[field] then s[field]=s[field]+elapsed end
            end
            s.path=nil s.jump=nil s.nextPlan=now
            return wait("Perjalanan vending dijeda saat karakter belum siap.")
        end
        if now-s.started>120 then return wait("Jalur vending terlalu lama; kembali farm.",true) end
        local function inside(x,y)
            return x>=0 and y>=0 and x<ctx.w and y<ctx.h
        end
        local function tile(x,y)
            return inside(x,y) and ctx.fg[y] and ctx.fg[y][x] or nil
        end
        local function body(x,y,startX,startY)
            local id=tile(x,y)
            local item=id and items[id]
            if id==nil or (id~=0 and not item) then return false end
            if item then
                if item.solid or item.hurts or item.deadly or item.bumper or item.laser or item.lock then return false end
                if (item.door or item.entrance or item.portal) and not (x==startX and y==startY) then return false end
            end
            return true
        end
        local function floorAt(x,y)
            local item=items[tile(x,y+1)]
            return item and (item.solid==true or item.platform==true)
                and not (item.hurts or item.deadly or item.bumper or item.laser
                    or item.door or item.entrance or item.portal or item.lock) or false
        end
        local machine=items[tile(vendor.x,vendor.y)]
        if not machine or not machine.vend then return wait("Vending tujuan sudah tidak ada.",true) end
        local x,feet=commerce.Position()
        if not finite(x) or not finite(feet) then releaseKeys() return wait("Menunggu posisi karakter.") end
        if not active() then api.Release() return wait("Perjalanan belanja dihentikan.") end
        -- Position may rediscover native UI/context. Never apply the old map's
        -- arrival or jump plan after that discovery has replaced its binding.
        if ctx~=commerce.Context or ctx.name~=target or shopWorld()~=target or game.JobId~=s.job then
            return wait("Sesi world berubah saat membaca posisi vending.",true)
        end
        local grounded=math.abs(feet-math.floor(feet+.5))<=.12
        local row,column=math.floor(feet+.5)-1,math.floor(x)
        if grounded and floorAt(column,row) and body(column,row)
            and ctx.atTile(vendor.x,vendor.y)==true then
            api.Release() return true,"Sudah dekat vending."
        end
        local function jumpClear(fromX,fromY,toX,toY,startX,startY)
            if math.abs(toX-fromX)~=1 or fromY-toY<1 or fromY-toY>2 or not floorAt(toX,toY) then return false end
            for y=toY-1,fromY do if not body(fromX,y,startX,startY) then return false end end
            return body(toX,toY) and body(toX,toY-1)
        end
        local function plan(startX,startY)
            if not body(startX,startY,startX,startY) or not floorAt(startX,startY) then return end
            local goals={}
            for _,dx in ipairs({0,-1,1}) do
                local gx,gy=vendor.x+dx,vendor.y
                if body(gx,gy) and floorAt(gx,gy) then
                    goals[gy*ctx.w+gx]={x=gx,y=gy,
                        px=dx==0 and gx+.5 or (dx<0 and gx+.85 or gx+.15)}
                end
            end
            local startKey=startY*ctx.w+startX
            local queue,parents,cursor={{x=startX,y=startY}},{[startKey]=false},1
            local endKey,goal
            while queue[cursor] and cursor<=math.min(ctx.w*ctx.h,8000) do
                local n=queue[cursor]
                local nk=n.y*ctx.w+n.x
                if goals[nk] then endKey,goal=nk,goals[nk] break end
                cursor=cursor+1
                local first=vendor.x>=n.x and 1 or -1
                for _,dir in ipairs({first,-first}) do
                    local nx,ny=n.x+dir,n.y
                    if body(nx,ny,startX,startY) then
                        local landing
                        if floorAt(nx,ny) then landing=ny
                        elseif tile(nx,ny+1)==0 or body(nx,ny+1,startX,startY) then
                            for below=ny+1,math.min(ny+4,ctx.h-2) do
                                if not body(nx,below,startX,startY) then break end
                                if floorAt(nx,below) then landing=below break end
                                local support=items[tile(nx,below+1)]
                                if support and (support.solid or support.platform) then break end
                            end
                        end
                        if landing then
                            local k=landing*ctx.w+nx
                            if parents[k]==nil and k~=startKey then
                                parents[k]={from=nk,fromX=n.x,fromY=n.y,x=nx,y=landing}
                                queue[#queue+1]={x=nx,y=landing}
                            end
                        end
                    end
                    if view.canSpaceJump then for rise=1,2 do
                        local landing=ny-rise
                        local k=landing*ctx.w+nx
                        if parents[k]==nil and k~=startKey and jumpClear(n.x,ny,nx,landing,startX,startY) then
                            parents[k]={from=nk,fromX=n.x,fromY=n.y,x=nx,y=landing,kind="jump"}
                            queue[#queue+1]={x=nx,y=landing}
                        end
                    end end
                end
            end
            if not endKey then return end
            local reverse,path={},{}
            while endKey~=startKey do
                local edge=parents[endKey]
                if not edge then return end
                reverse[#reverse+1]=edge endKey=edge.from
            end
            for i=#reverse,1,-1 do path[#path+1]=reverse[i] end
            if #path==0 then path[1]={x=goal.x,y=goal.y,fromX=goal.x,fromY=goal.y} end
            path[#path].targetX=goal.px
            return path
        end
        local function replan(message)
            releaseKeys() s.path=nil s.jump=nil s.nextPlan=now+.5
            return wait(message)
        end
        if not s.path then
            releaseKeys()
            if not grounded then return wait("Menunggu mendarat sebelum menuju vending.") end
            if now<s.nextPlan then return wait("Membaca ulang jalur vending.") end
            s.nextPlan=now+1
            local path=plan(column,row)
            if not path then
                s.blockedAt=s.blockedAt or now
                return wait("Jalur aman ke vending belum tersedia.",now-s.blockedAt>=10)
            end
            s.blockedAt=nil s.path=path s.index=1 s.startX=column s.startY=row
            s.progressX=x s.progressY=feet s.progressAt=now s.legAt=now
        end
        local edge=s.path[s.index]
        local targetX=edge.targetX or edge.x+.5
        local tolerance=edge.targetX and .06 or .2
        if math.abs(x-s.progressX)>.12 or math.abs(feet-s.progressY)>.12 then
            s.progressX=x s.progressY=feet s.progressAt=now
        end
        if now-s.progressAt>4 or now-s.legAt>15 then return replan("Jalur vending terhalang; membaca ulang.") end
        if grounded and row==edge.y and column==edge.x and math.abs(x-targetX)<=tolerance then
            releaseKeys() s.index=s.index+1 s.jump=nil
            if not s.path[s.index] then s.path=nil s.nextPlan=now end
            s.progressX=x s.progressY=feet s.progressAt=now s.legAt=now
            return wait("Melanjutkan jalan menuju vending.")
        end
        if edge.kind=="jump" then
            if not view.canSpaceJump or not jumpClear(edge.fromX,edge.fromY,edge.x,edge.y,s.startX,s.startY)
                or (column~=edge.fromX and column~=edge.x) then return replan("Ruang lompat ke vending berubah.") end
            local jumpKey=edge.fromX..":"..edge.fromY..":"..edge.x..":"..edge.y
            s.attempts=s.attempts or {}
            if not s.jump then
                if not grounded or row~=edge.fromY or column~=edge.fromX then return replan("Menunggu pijakan lompat siap.") end
                if (s.attempts[jumpKey] or 0)>=2 then return wait("Undakan vending tidak terjangkau; kembali farm.",true) end
                local dx=edge.fromX+.5-x
                if math.abs(dx)>.15 then
                    key(Enum.KeyCode.Space,false) key(Enum.KeyCode.Left,dx<0) key(Enum.KeyCode.Right,dx>0)
                    return wait("Menempatkan karakter sebelum lompat.")
                end
                releaseKeys() s.attempts[jumpKey]=(s.attempts[jumpKey] or 0)+1
                s.jump={started=now,releaseAt=now+.2,crossed=false}
                key(Enum.KeyCode.Space,true)
                return wait("Lompat normal menuju vending.")
            end
            local jump=s.jump
            if now>=jump.releaseAt then key(Enum.KeyCode.Space,false) end
            if now-jump.started>2.5 or (now-jump.started>.45 and grounded and row>=edge.fromY) then
                return replan("Lompatan belum mencapai undakan vending.")
            end
            if feet<=edge.y+1-.05 then jump.crossed=true end
            local dx=targetX-x
            key(Enum.KeyCode.Left,jump.crossed and dx<-.15)
            key(Enum.KeyCode.Right,jump.crossed and dx>.15)
            return wait("Melewati undakan menuju vending.")
        end
        local valid=floorAt(edge.x,edge.y) and (column==edge.fromX or column==edge.x)
        for y=edge.fromY,edge.y do
            if not body(edge.x,y,s.startX,s.startY) then valid=false break end
            local support=items[tile(edge.x,y+1)]
            if y<edge.y and support and (support.solid or support.platform) then valid=false break end
        end
        if grounded then
            if not body(column,row,s.startX,s.startY) then valid=false end
            local dropping=edge.y>edge.fromY and column==edge.x and row>=edge.fromY and row<edge.y
            if not dropping and not floorAt(column,row) then valid=false end
            if row<edge.fromY or row>edge.y then valid=false end
        end
        if not valid then return replan("Jalur vending berubah; menunggu jalur aman.") end
        if not grounded and edge.y==edge.fromY then return replan("Menunggu mendarat di jalur vending.") end
        if not active() or ctx~=commerce.Context or ctx.name~=target or shopWorld()~=target or game.JobId~=s.job then
            api.Release() return wait("Perjalanan vending dibatalkan.")
        end
        local dx=targetX-x
        key(Enum.KeyCode.Space,false)
        key(Enum.KeyCode.Left,dx<-tolerance) key(Enum.KeyCode.Right,dx>tolerance)
        return wait(edge.y>edge.fromY and "Turun lewat tepi menuju vending." or "Berjalan menuju vending.")
    end
    function api.Step(vendor,now)
        local ok,arrived,message,fatal=pcall(step,vendor,now)
        if not ok then api.Release() return false,"Kontrol perjalanan vending belum siap: "..tostring(arrived):sub(1,100),true end
        return arrived,message,fatal
    end
    return api
end

end)()

local makeShopping = (function()
-- Serialized, account-local commerce. Native handlers own all inventory changes.
-- Step never waits for native IO; workers report into the shared pending record.
return function(options)
    options=type(options)=="table" and options or {}
    local io=options.io or {}
    local MAX=9007199254740991
    local function integer(v,lo,hi)
        return type(v)=="number" and v==v and v%1==0 and v>=(lo or 0) and v<=(hi or MAX)
    end
    local function world(v) return type(v)=="string" and #v>0 and #v<=24 and v:match("^[A-Z0-9]+$")~=nil end
    local function copy(v)
        if type(v)~="table" then return v end
        local out={} for k,x in pairs(v) do out[k]=copy(x) end return out
    end
    local function equal(a,b)
        if type(a)~=type(b) then return false end
        if type(a)~="table" then return a==b end
        for k,v in pairs(a) do if not equal(v,b[k]) then return false end end
        for k in pairs(b) do if a[k]==nil then return false end end
        return true
    end
    local shared=type(options.pending)=="table" and options.pending or {}
    local owner={} shared.owner=owner
    local alive,running,stepping=true,false,false
    local configuredWorld=options.shopWorld==nil and "BLOMM" or options.shopWorld
    local state={phase="idle",budget=0,protectedWL=0,sequence=0,shopWorld=configuredWorld}
    local revision,disabled,message=0,nil,"Menunggu gems untuk belanja item vending."
    local nextTry,receiptSince,vendorSince,travelSince,walletSince,itemSince=0,nil,nil,nil,nil,nil
    local cooldown=integer(options.cooldown,1,3600) and options.cooldown or 60
    local receiptTimeout=integer(options.receiptTimeout,1,300) and options.receiptTimeout or 20
    local vendorTimeout=integer(options.vendorTimeout,1,300) and options.vendorTimeout or 15
    local travelTimeout=integer(options.travelTimeout,1,600) and options.travelTimeout or 120
    local walletTimeout=integer(options.walletTimeout,1,300) and options.walletTimeout or 20
    local cooldownLoaded=false
    local identity=integer(options.userId,1) and integer(options.placeId,1)
    local prefix=identity and ("BLUE_HUB_GRASS_SHOP_"..string.format("%.0f",options.userId).."_"..string.format("%.0f",options.placeId)) or nil
    local paths=prefix and {prefix.."_a.json",prefix.."_b.json"} or {}
    local phases={idle=true,buy_wl=true,outbound=true,vending=true,returning=true,cooldown=true,blocked=true}
    local api={Path=paths[1],BackupPath=paths[2]}
    local function current() return alive and shared.owner==owner end
    local function release()
        if type(io.ReleaseMovement)=="function" then pcall(io.ReleaseMovement) end
    end
    local function fail(why)
        if current() then release() end
        disabled=tostring(why) message="Belanja nonaktif: "..disabled
        return false,message
    end
    local function vendorPoint(key)
        if type(key)~="string" then return end
        local name,x,y=key:match("^([A-Z0-9]+):(%d+):(%d+)$")
        x,y=tonumber(x),tonumber(y)
        if world(name) and integer(x,0,4095) and integer(y,0,4095)
            and key==string.format("%s:%d:%d",name,x,y) then return x,y,name end
    end
    local function vendorWorld(key)
        local _,_,name=vendorPoint(key)
        return name
    end
    local function journalWorld(s)
        return s.shopWorld or vendorWorld(s.vendorKey)
            or s.pending and vendorWorld(s.pending.vendorKey) or "BLOMM"
    end
    local function valid(raw)
        if type(raw)~="table" or raw.schema~="BLUE_HUB_GRASS_SHOP" or raw.version~=1
            or raw.userId~=options.userId or raw.placeId~=options.placeId or not integer(raw.revision,1) then return false end
        local s=raw.state
        if type(s)~="table" or not phases[s.phase] or not integer(s.budget) or not integer(s.protectedWL)
            or not integer(s.sequence) or (s.home~=nil and not world(s.home))
            or (s.shopWorld~=nil and not world(s.shopWorld))
            or (s.vendorKey~=nil and (vendorPoint(s.vendorKey)==nil or vendorWorld(s.vendorKey)~=journalWorld(s)))
            or (s.spendAllWL~=nil and type(s.spendAllWL)~="boolean")
            or (s.currencyRemaining~=nil and not integer(s.currencyRemaining))
            or (s.lastReturnReason~=nil and (type(s.lastReturnReason)~="string" or #s.lastReturnReason>500)) then return false end
        if s.phase~="idle" and s.phase~="blocked" and not world(s.home) then return false end
        if s.pending~=nil then
            local p=s.pending local b=type(p)=="table" and p.before
            if type(p)~="table" or not integer(p.id,1) or p.id>s.sequence or type(b)~="table"
                or not integer(b.gems) or not integer(b.wl)
                or not integer(p.qty,1) or not integer(p.price,1) then return false end
            if p.kind=="buy_wl" then
                if p.qty~=1 or s.phase~="buy_wl" and s.phase~="blocked" then return false end
            elseif p.kind=="buy_grass" then
                if p.price~=1 or type(p.vendorKey)~="string" or #p.vendorKey==0 or s.budget<p.qty
                    or not integer(b.grass) or s.phase~="vending" and s.phase~="blocked" then return false end
            elseif p.kind=="buy_item" then
                if p.price~=1 or vendorPoint(p.vendorKey)==nil or vendorWorld(p.vendorKey)~=journalWorld(s) or s.budget<p.qty
                    or not integer(p.itemId,1,2147483647) or not integer(b.itemCount)
                    or s.vendorKey~=nil and s.vendorKey~=p.vendorKey
                    or s.phase~="vending" and s.phase~="blocked" then return false end
            else return false end
        end
        return true
    end
    local function absent(err)
        local s=tostring(err):lower()
        return s:find("no such file",1,true) or s:find("not found",1,true) or s:find("cannot find",1,true)
            or s:find("does not exist",1,true) or s:find("doesn't exist",1,true)
    end
    local function envelope(s,rev)
        return {schema="BLUE_HUB_GRASS_SHOP",version=1,userId=options.userId,placeId=options.placeId,revision=rev,state=s}
    end
    local function save(candidate)
        if not current() or disabled then return false end
        local raw=envelope(copy(candidate or state),revision+1)
        if not valid(raw) then return fail("Journal transaksi tidak valid.") end
        local ok,encoded=pcall(options.encode,raw)
        if not current() then return false end
        if not ok or type(encoded)~="string" then return fail("Journal tidak dapat dikodekan.") end
        for _,path in ipairs(paths) do
            if not current() then return false end
            local wrote,result=pcall(options.write,path,encoded)
            if not current() then return false end
            if not wrote or result==false then return fail("Journal tidak dapat disimpan; tidak ada pembelian baru.") end
            local readOK,back=pcall(options.read,path)
            if not current() then return false end
            if not readOK or back~=encoded then return fail("Verifikasi journal gagal; tidak ada pembelian baru.") end
        end
        state=raw.state revision=raw.revision
        return true
    end
    if not world(configuredWorld) then fail("VendingWorld harus berisi 1-24 huruf A-Z atau angka.") end
    if not identity then fail("Identitas akun belum tersedia.") end
    for _,name in ipairs({"read","write","encode","decode","spawn"}) do
        if type(options[name])~="function" then fail("Penyimpanan lokal/spawn tidak tersedia.") end
    end
    if not disabled then
        local loaded={}
        for _,path in ipairs(paths) do
            local ok,raw=pcall(options.read,path)
            if ok and raw~=nil then
                local decoded,value=pcall(options.decode,raw)
                if not decoded or not valid(value) then fail("Journal rusak atau identitas berbeda; periksa saldo.") break end
                loaded[#loaded+1]=value
            elseif not ok and not absent(raw) then fail("Journal tidak dapat dibaca; periksa izin file.") break end
        end
        if not disabled and #loaded>0 then
            table.sort(loaded,function(a,b) return a.revision>b.revision end)
            if #loaded==2 and loaded[1].revision==loaded[2].revision then
                if not equal(loaded[1],loaded[2]) then fail("Salinan journal berbeda; periksa saldo.") end
            end
            if not disabled then state=copy(loaded[1].state) revision=loaded[1].revision end
        end
        -- Older journals pinned only the pending Grass transaction. Preserve
        -- that known machine when its receipt is cleared during migration.
        if not disabled and state.vendorKey==nil and state.pending and vendorPoint(state.pending.vendorKey)~=nil then
            state.vendorKey=state.pending.vendorKey
        end
        -- Older files did not record the destination separately. Preserve the
        -- original machine/receipt world until reconciliation finishes.
        if not disabled then state.shopWorld=journalWorld(state) end
        cooldownLoaded=state.phase=="cooldown"
    end
    local function read()
        if type(io.Read)~="function" then return {ready=false,message="Saldo belum tersedia."} end
        local ok,v=pcall(io.Read)
        return ok and type(v)=="table" and v or {ready=false,message="Pembacaan saldo gagal."}
    end
    local function wallet(v)
        return v.ready==true and integer(v.gems) and integer(v.wl)
            and integer(v.wlRoom) and integer(v.price,1)
    end
    local function readItem(id,legacyWallet)
        local ok,v,why
        if type(io.ReadItem)=="function" then ok,v,why=pcall(io.ReadItem,id)
        elseif id==16 and legacyWallet and integer(legacyWallet.grass) and integer(legacyWallet.grassRoom) then
            ok,v=true,{itemId=16,count=legacyWallet.grass,room=legacyWallet.grassRoom,name="Grass"}
        end
        if ok and type(v)=="table" and v.ready~=false and v.itemId==id and integer(v.count) and integer(v.room) then return v end
        return nil,type(why)=="string" and why or "Menunggu data item di tas."
    end
    local function activeOperation() return type(shared.operation)=="table" and not shared.operation.done end
    local function launch(kind,fn,now,txId,prepared)
        if not current() or not running or (activeOperation() and shared.operation~=prepared) or type(fn)~="function" then return false end
        local op=prepared or {kind=kind,txId=txId,done=false,started=false,nativePending=true,at=now,owner=owner}
        shared.operation=op
        local ok,why=pcall(options.spawn,function()
            if not current() or not running then
                op.cancelled=true op.done=true op.nativePending=false return
            end
            op.started=true
            local called,a,b,c,d=pcall(fn)
            -- This is deliberately the only write made after a native call:
            -- a replacement coordinator can safely consume the same receipt.
            op.called,op.accepted,op.detail,op.value,op.uncertain=called,a,b,c,d==true or not called
            op.done=true op.nativePending=false
        end)
        if not ok then op.cancelled=true op.detail=tostring(why) op.done=true op.nativePending=false end
        return true
    end
    local function absorb()
        local op=shared.operation
        if type(op)~="table" or not op.done then return end
        if op.kind=="buy_wl" or op.kind=="buy_grass" or op.kind=="buy_item" then
            shared.receipt=op
        elseif op.kind=="move" and op.called and op.value==true and not op.accepted then
            shared.moveFatal=op.detail or "Jalur vending tidak tersedia."
        end
        shared.operation=nil
    end
    local function change(phase,why)
        release()
        local s=copy(state) s.phase=phase
        if phase=="returning" or phase=="cooldown" and (state.phase~="returning" or not s.lastReturnReason) then
            s.lastReturnReason=why or message
        end
        if not save(s) then return false end
        message=why or message vendorSince=nil travelSince=nil itemSince=nil
        return true
    end
    local function block(why)
        release()
        local s=copy(state) s.phase="blocked" s.reason=why
        save(s) message=why
        return false,message
    end
    local function callTravel(target,now)
        if not world(target) then message="Menunggu nama My World akun ini." return end
        if not travelSince then travelSince=now end
        if now-travelSince>=travelTimeout then block("Perjalanan belanja tertahan; periksa world dan saldo.") return end
        if now<nextTry then return end
        nextTry=now+.3
        launch("travel",function() return io.Travel(target) end,now)
    end
    local function reconcile(v,now)
        release()
        local p=state.pending
        if not p then shared.receipt=nil receiptSince=nil return true end
        local op=shared.operation
        if op and not op.done then message="Menunggu hasil transaksi; tidak mengirim ulang." return false end
        local r=shared.receipt
        if r and r.txId~=p.id then block("Identitas transaksi tidak cocok; periksa saldo.") return false end
        if r and r.cancelled and not r.started then
            local s=copy(state) s.pending=nil
            if save(s) then shared.receipt=nil receiptSince=nil end
            message="Pembelian dibatalkan sebelum dikirim." return false
        end
        if not wallet(v) then
            message=v.message or "Menunggu saldo terbaru untuk memeriksa transaksi."
            local expected=p.kind~="buy_wl" and (vendorWorld(p.vendorKey) or state.shopWorld) or state.home
            local needsWorld=v.picker==true or v.world=="" or world(v.world) and v.world~=expected
            -- Cache refresh after a native receipt is normal. Opening the
            -- picker merely because gems are not fresh would leave the vendor.
            if running and not activeOperation() and needsWorld then
                callTravel(expected,now)
            end
            return false
        end
        receiptSince=receiptSince or now
        local b=p.before
        local itemId=p.kind=="buy_grass" and 16 or p.itemId
        local item,beforeItem
        if p.kind~="buy_wl" then
            item,message=readItem(itemId,p.kind=="buy_grass" and v or nil)
            if not current() then return false end
            if not item then
                if now-receiptSince>=receiptTimeout then block("Data item transaksi belum tersedia; belanja dihentikan tanpa mengirim ulang.") end
                return false
            end
            beforeItem=p.kind=="buy_grass" and b.grass or b.itemCount
        end
        local nativeWLReceipt=r and r.called and r.accepted==true and not r.uncertain
        local confirmed=p.kind=="buy_wl" and v.wl==b.wl+1 and (nativeWLReceipt or v.gems==b.gems-p.price)
            or p.kind~="buy_wl" and itemId~=242 and v.wl==b.wl-p.qty and item.count==beforeItem+p.qty
            or p.kind=="buy_item" and itemId==242 and nativeWLReceipt and v.wl==b.wl and item.count==beforeItem
        if confirmed then
            local s=copy(state) s.pending=nil
            s.budget=v.wl s.protectedWL=0
            if p.kind=="buy_item" and itemId==242 then
                s.currencyRemaining=math.max(0,(s.currencyRemaining or b.wl)-p.qty)
            end
            if save(s) then
                shared.receipt=nil receiptSince=nil
                message=p.kind=="buy_wl" and "World Lock baru terkonfirmasi." or "Pembelian item vending terkonfirmasi."
                return true
            end
            return false
        end
        local unchanged=v.wl==b.wl and (p.kind=="buy_wl" and v.gems==b.gems or p.kind~="buy_wl" and item.count==beforeItem)
        if r and r.called and r.accepted==false and not r.uncertain and unchanged then
            local s=copy(state) s.pending=nil
            s.budget=v.wl s.protectedWL=0
            s.phase=p.kind=="buy_wl" and (s.budget>0 and "outbound" or "cooldown") or "returning"
            if s.phase=="returning" or s.phase=="cooldown" then
                s.lastReturnReason="Pembelian ditolak; kembali farm tanpa mengulang transaksi."
            end
            if save(s) then
                shared.receipt=nil receiptSince=nil nextTry=s.phase=="cooldown" and now+cooldown or 0
                message="Pembelian ditolak; kembali farm tanpa mengulang transaksi."
            end
            return false
        end
        if now-receiptSince>=receiptTimeout then
            block("Hasil pembelian belum pasti; belanja dihentikan agar tidak membeli dua kali.")
        else message="Memeriksa perubahan saldo; tidak mengirim ulang transaksi." end
        return false
    end
    local function prepare(kind,v,qty,price,vendor,now,item)
        if not current() or not running then return false end
        release()
        if state.pending or activeOperation() then return false end
        local s=copy(state) s.sequence=s.sequence+1
        s.pending={id=s.sequence,kind=kind,qty=qty,price=price,
            before={gems=v.gems,wl=v.wl,grass=v.grass,itemCount=item and item.count or nil},
            itemId=vendor and vendor.itemId or nil,vendorKey=vendor and vendor.key or nil}
        -- Record the not-yet-sent operation before file callbacks can yield.
        -- STOP/reload during persistence can then prove this buy was unsent.
        local op={kind=kind,txId=s.sequence,done=false,started=false,nativePending=true,at=now,owner=owner}
        shared.operation=op
        local function cancel() op.cancelled=true op.done=true op.nativePending=false return false end
        if not save(s) then return cancel() end
        local id=state.pending.id receiptSince=nil
        local fn=kind=="buy_wl" and function() return io.BuyWL() end or function() return io.BuyItem(vendor,qty) end
        if not launch(kind,fn,now,id,op) then return cancel() end
        return true
    end
    local function quote(v)
        if type(v)~="table" or not integer(v.itemId,1,2147483647) or v.cost~=1 or v.per~=1 or not integer(v.stock,1) then return false end
        local x,y,name=vendorPoint(v.key)
        return x~=nil and name==state.shopWorld and v.x==x and v.y==y and (state.vendorKey==nil or v.key==state.vendorKey)
    end
    function api.Pending() return shared end
    function api.Active()
        if not current() then return false end
        if activeOperation() then return true end
        if disabled then return false end
        if not state.pending and state.shopWorld~=configuredWorld then return true end
        if state.phase=="blocked" then return false end
        return walletSince~=nil or state.pending~=nil or state.phase=="buy_wl" or state.phase=="outbound"
            or state.phase=="vending" or state.phase=="returning"
    end
    function api.Status()
        return {phase=walletSince and "waiting_wallet" or state.phase,budget=state.budget,protectedWL=state.protectedWL,
            pending=copy(state.pending),disabled=disabled,message=message,nextTry=nextTry,revision=revision,vendorKey=state.vendorKey,
            currencyRemaining=state.currencyRemaining,lastReturnReason=state.lastReturnReason,spendAllWL=state.spendAllWL,
            shopWorld=state.shopWorld,configuredShopWorld=configuredWorld}
    end
    local function eligible(v,now)
        if cooldownLoaded then nextTry=now+cooldown cooldownLoaded=false end
        if now<nextTry then return false,"Menunggu putaran belanja berikutnya." end
        if not wallet(v) or not world(v.homeWorld) then return false,v.message or "Menunggu saldo dan My World.",true end
        if v.world~=v.homeWorld then return false,"Farm harus berada di My World sebelum belanja." end
        return v.wl>0 or v.gems>=v.price and v.wlRoom>0,"World Lock "..v.wl.."; gems "..v.gems.." / "..v.price.."."
    end
    function api.Ready(now)
        now=tonumber(now) or 0
        if not current() then return false,"Belanja ditutup." end
        if disabled then return false,message end
        if state.phase=="blocked" and (state.pending or state.shopWorld==configuredWorld) then return false,state.reason or "Belanja menunggu pemeriksaan saldo." end
        if api.Active() then return true,message end
        local v=read()
        if not current() then return false,"Belanja ditutup." end
        return eligible(v,now)
    end
    local function step(now,wantsRun)
        if not current() then return false,"Belanja ditutup.",state.phase end
        running=wantsRun==true
        if not running then walletSince=nil end
        absorb()
        if disabled then return activeOperation(),message,state.phase end
        if state.phase=="blocked" and (state.pending or state.shopWorld==configuredWorld) then return activeOperation(),state.reason or message,state.phase end
        local v=read()
        if not current() then return false,"Belanja ditutup.",state.phase end
        if state.pending then
            if not reconcile(v,now) then return api.Active(),message,state.phase end
            -- Re-read on the next heartbeat after accounting; never use an old
            -- receipt snapshot as the baseline of the following transaction.
            return api.Active(),message,state.phase
        end
        if activeOperation() then return true,message,state.phase end
        if not running then release() return false,"Belanja dihentikan.",state.phase end
        if state.shopWorld~=configuredWorld then
            -- Never reinterpret a saved machine coordinate under a new world.
            -- Pending receipts and native workers were settled above first.
            release()
            local s=copy(state)
            s.shopWorld=configuredWorld s.vendorKey=nil s.currencyRemaining=nil
            s.reason=nil s.lastReturnReason=nil
            if s.phase=="outbound" or s.phase=="vending" or s.phase=="returning" or s.phase=="blocked" then
                s.phase=world(s.home) and (wallet(v) and v.wl==0 and "returning" or "outbound") or "idle"
            elseif s.phase=="cooldown" then s.phase="idle" end
            if not save(s) then return false,message,state.phase end
            shared.moveFatal=nil vendorSince=nil travelSince=nil itemSince=nil walletSince=nil
            nextTry=0 cooldownLoaded=false
            message="World vending diubah ke "..configuredWorld.."; memeriksa tujuan baru."
            return api.Active(),message,state.phase
        end
        if state.spendAllWL~=true and state.phase=="returning" and v.world==state.shopWorld and not wallet(v) then
            release() walletSince=walletSince or now
            message=v.message or "Memeriksa sisa World Lock sebelum kembali."
            if now-walletSince<walletTimeout then return true,message,"waiting_wallet" end
            walletSince=nil
            change("cooldown","Saldo belum siap untuk memeriksa sisa WL; belanja ditunda.")
            nextTry=now+cooldown
            return false,message,state.phase
        end
        if wallet(v) then walletSince=nil end
        -- This policy now spends every WL in the backpack. Migrate the old
        -- protected/new-only budget only after any pending receipt is settled.
        if wallet(v) and (state.spendAllWL~=true or state.protectedWL~=0 or state.budget~=v.wl) then
            local s=copy(state) s.protectedWL=0 s.budget=v.wl s.spendAllWL=true
            -- Old policy could already be returning with a protected WL left.
            -- Re-evaluate once in the shop world, then persist the new policy
            -- so a genuine stock/path failure is not retried on every reload.
            if state.spendAllWL~=true and state.phase=="returning" and v.world==state.shopWorld and v.wl>0 then
                s.phase="vending"
            end
            if not save(s) then return false,message,state.phase end
        end
        if state.phase=="idle" or state.phase=="cooldown" then
            -- The runtime pauses farming and refreshes the wallet after Ready.
            -- A pending refresh is not evidence that shopping became ineligible.
            -- Keep that pause until the fresh snapshot arrives; never use a
            -- second Read whose readiness could differ from this baseline.
            local ready,why,pendingWallet=eligible(v,now)
            if not ready then
                message=why release()
                local needsRecovery=v.picker==true or v.loading==true or v.sessionReady==false or v.world==""
                    or world(v.world) and world(v.homeWorld) and v.world~=v.homeWorld
                if pendingWallet and not needsRecovery then
                    walletSince=walletSince or now
                    if now-walletSince<walletTimeout then return true,message,"waiting_wallet" end
                    nextTry=now+cooldown
                    message="Saldo gems belum siap; lanjut farm dan coba belanja lagi nanti."
                end
                walletSince=nil
                return false,message,state.phase
            end
            walletSince=nil
            local s=copy(state) s.home=v.homeWorld
            s.protectedWL=0 s.budget=v.wl s.phase="buy_wl" s.currencyRemaining=nil
            if not save(s) then return false,message,state.phase end
            message="Menyiapkan semua World Lock di tas untuk belanja."
            return true,message,state.phase
        end
        if state.phase=="outbound" or state.phase=="returning" then
            local target=state.phase=="outbound" and state.shopWorld or (world(v.homeWorld) and v.homeWorld or state.home)
            if v.world==target and v.ready then
                travelSince=nil nextTry=0
                if state.phase=="outbound" then change("vending","Mencari item vending: 1 WL per item.")
                else
                    if change("cooldown","Kembali ke My World; melanjutkan farm.") then nextTry=now+cooldown end
                    return false,message,state.phase
                end
            else message="Menuju "..tostring(target).."." callTravel(target,now) end
            return api.Active(),message,state.phase
        end
        if not wallet(v) then release() return true,v.message or "Menunggu saldo terbaru.",state.phase end
        if state.phase=="buy_wl" then
            if v.world~=state.home then change("returning","World berubah; kembali My World.")
            elseif v.gems>=v.price and v.wlRoom>0 then
                message="Membeli World Lock dengan gems."
                prepare("buy_wl",v,1,v.price,nil,now)
            elseif state.budget>0 then change("outbound","World Lock siap; menuju "..state.shopWorld..".")
            else change("cooldown","Gems belum cukup; lanjut farm.") nextTry=now+cooldown end
        elseif state.phase=="vending" then
            if v.world~=state.shopWorld then change("outbound","Kembali ke world vending.") return true,message,state.phase end
            if shared.moveFatal then
                local why=tostring(shared.moveFatal)
                shared.moveFatal=nil change("returning",why)
                return true,message,state.phase
            end
            local available=v.wl
            if available<1 then change("returning","World Lock di tas habis; kembali My World.")
            else
                local ok,vendor,why=pcall(io.FindVendor,state.vendorKey)
                if not current() then return false,"Belanja ditutup.",state.phase end
                if not ok or not quote(vendor) then
                    release()
                    vendorSince=vendorSince or now
                    message=not ok and tostring(vendor) or (type(why)=="string" and why or "Menunggu mesin vending yang sama: tepat 1 WL per item.")
                    if now-vendorSince>=vendorTimeout then change("returning",message) end
                else
                    vendorSince=nil
                    if not state.vendorKey then
                        local s=copy(state) s.vendorKey=vendor.key
                        if not save(s) then return false,message,state.phase end
                    end
                    local item,why=readItem(vendor.itemId)
                    if not current() then return false,"Belanja ditutup.",state.phase end
                    if not item then
                        release() itemSince=itemSince or now
                        if now-itemSince>=vendorTimeout then
                            change("returning","Data item vending belum tersedia; kembali farm.")
                            return true,message,state.phase
                        end
                        return true,why,state.phase
                    end
                    itemSince=nil
                    local name=type(item.name)=="string" and item.name or ("item "..vendor.itemId)
                    if vendor.itemId==242 then
                        if state.currencyRemaining==nil then
                            local s=copy(state) s.currencyRemaining=v.wl
                            if not save(s) then return false,message,state.phase end
                        end
                        available=math.min(available,state.currencyRemaining)
                        if available<1 then
                            change("returning","Vending berisi World Lock; tukar WL ke WL tidak mengurangi saldo. Kembali farm.")
                            return true,message,state.phase
                        end
                    end
                    if item.room<1 then
                        change("returning","Tas "..name.." penuh; kembali My World.")
                        return true,message,state.phase
                    end
                    local nearOK,near=pcall(io.AtVendor,vendor)
                    if not nearOK or not near then
                        message="Berjalan ke mesin vending."
                        launch("move",function() return io.MoveVendor(vendor) end,now)
                    else
                        local qty=math.min(available,vendor.stock,item.room)
                        message="Membeli "..qty.." "..name.." dengan World Lock di tas."
                        prepare("buy_item",v,qty,1,vendor,now,item)
                    end
                end
            end
        end
        return api.Active(),message,state.phase
    end
    function api.Step(now,wantsRun)
        if stepping then return api.Active(),message,state.phase end
        stepping=true
        local ok,busy,why,phase=pcall(step,tonumber(now) or 0,wantsRun)
        stepping=false
        if not ok then
            fail("Koordinator belanja gagal: "..tostring(busy):sub(1,120))
            return activeOperation(),message,state.phase
        end
        return busy,why,phase
    end
    function api.Stop() running=false walletSince=nil itemSince=nil if current() then release() end end
    function api.Close() running=false walletSince=nil if current() then release() end alive=false end
    return api
end

local function equal(a,b)
    if type(a)~=type(b) then return false end
    if type(a)~="table" then return a==b end
    for k,v in pairs(a) do if not equal(v,b[k]) then return false end end

    return true

recoverStep=function(now,view)
    if not app.WantsRun then return false end
    if not app.Recovering then
        if view.picker or not view.ready or view.loading or view.world==""
            or view.world~=app.HomeWorld or worldChanged() then beginRecovery(now)
        else return false end
    end
    local plan,homeMessage=ensureHomePlan()
    if not plan then
        if adapter then adapter.Release() end
        status(homeMessage or "Menunggu My World akun ini.") return true
    end
    if view.picker or not view.ready or view.loading or view.world=="" or view.world~=plan.world then
        recoveryBound=false arrivalAt=nil
        if adapter then adapter.Release() end
        if not view.ready or view.loading or view.typing then
            status(view.typing and "Pemulihan dijeda saat mengetik." or "Menunggu game siap untuk masuk My World.")
        elseif joinPending then status("Sedang masuk My World: "..plan.world.."...")
        elseif now<nextJoin then status("Masuk My World: "..plan.world.." dalam "..math.ceil(nextJoin-now).." detik.")
        else
            local token,target=recoveryToken,plan.world
            joinPending=true joinAttempts=joinAttempts+1
            status("Masuk My World: "..target.."...")
            task.spawn(function()
                if not app.Alive or not app.WantsRun or token~=recoveryToken then return end
                local ok,entered,message
                if not view.picker then ok,entered,message=pcall(session.OpenPicker)
                else ok,entered,message=pcall(session.Enter,target) end
                if not app.Alive or not app.WantsRun or token~=recoveryToken then return end
                joinPending=false
                nextJoin=os.clock()+(not view.picker and ok and entered~=false and .25
                    or math.min(30,5*2^math.min(joinAttempts-1,3)))
                if not ok or entered==false then status("Belum bisa masuk "..target..": "..tostring(ok and message or entered):sub(1,110).."; mencoba lagi.") end
            end)
        end
        return true
    end
    if not recoveryBound then
        if now<nextBind then return true end
        nextBind=now+1
        local called,ready,message,fatal=pcall(restorePlan,plan)
        if not called or not ready then
            if adapter then adapter.Release() end
            if called and fatal then
                rejectRestore(message)
            else status("Menunggu data / izin world: "..tostring(called and message or ready):sub(1,130)) end
            return true
        end
        recoveryBound=true
    end
    local ok,arrived,message,fatal=pcall(adapter.ReturnToAnchor,now)
    if not ok then adapter.Release() arrivalAt=nil status("Menunggu jalur kembali: "..tostring(arrived):sub(1,130))
    elseif fatal then stopRun(true) status(tostring(message or "Periksa jalur / izin titik tetap X 36 / Y 36."))
    elseif arrived then
        arrivalAt=arrivalAt or now
        status("Sudah di titik farm. Menunggu posisi stabil...")
        if now-arrivalAt>=.4 then
            adapter.Release() stopOtherController() engine.Start()
            app.Recovering=false app.RestoredPoint=false app.WantsRun=engine.State.running==true
            saveConfig(true) status(engine.State.status)
        end
    else arrivalAt=nil status(message or "Berjalan ke titik tetap X 36 / Y 36...") end
    return true
end

local function loadSettings()
    local http=game:GetService("HttpService")
    config=makeConfigStore({userId=player.UserId,placeId=game.PlaceId,read=readfile,write=writefile,
        encode=function(v) return http:JSONEncode(v) end,decode=function(v) return http:JSONDecode(v) end})
    app.ConfigStore=config
    local settings,message=config.Load()
    settings=type(settings)=="table" and settings or {}
    app.Mode=validMode(savedMode) and savedMode or (validMode(settings.mode) and settings.mode or "farm")
    app.PlantCount=validPlantCount(savedPlantCount) and savedPlantCount or (validPlantCount(settings.plantCount) and settings.plantCount or 21)
    selectedSeedId=savedSeedId or settings.seedId or 17
    local plan
    if hadLive then
        plan=savedRun
        if not plan and savedAnchor and (savedAnchor.mode or "farm")==app.Mode and (savedAnchor.plantCount or 21)==app.PlantCount then
            plan={world=savedAnchor.world,anchor=savedAnchor,seedId=selectedSeedId,mode=app.Mode,plantCount=app.PlantCount,progress=savedProgress}
        end
        if plan then
            plan=copy(plan) plan.mode=plan.mode or app.Mode plan.plantCount=plan.plantCount or app.PlantCount
            if plan.mode~=app.Mode or plan.plantCount~=app.PlantCount or plan.seedId~=selectedSeedId then plan=nil
            elseif savedProgress then plan.progress=copy(savedProgress) end
        end
    else plan=copy(settings.lastRun) end
    app.PendingPlan=plan app.SavedPlan=nil app.LastRun=nil
    app.NeedsCapture=false app.RestoredPoint=true
    settingsLoaded=true app.SettingsLoaded=true app.UserId=player.UserId
    ui.SetAnchor(nil)
    configStatus(message,nil)
    if not cancelInitialResume then app.Start() end
    if cancelInitialResume then saveConfig(true) end
end

    status("Titik tetap X 36 / Y 36 di My World siap. START untuk farm.")
    return true

    function io.BuyWL() syncShoppingWorld() return commerce.BuyWL() end
    function io.FindVendor(targetKey) syncShoppingWorld() return commerce.FindVendor(targetKey) end
    function io.ReadItem(itemId) syncShoppingWorld() return commerce.ReadItem(itemId) end
    function io.AtVendor(v) syncShoppingWorld() return commerce.NearVendor(v) end
    function io.MoveVendor(v) syncShoppingWorld() return shopWalk.Step(v,os.clock()) end
    function io.ReleaseMovement()
        if type(shopWalk.Pause)=="function" then shopWalk.Pause() else shopWalk.Release() end
    end
    function io.BuyItem(v,quantity) syncShoppingWorld() return commerce.BuyItem(v,quantity) end
    function io.Travel(world)
        if not app.Alive or not app.WantsRun or not app.Shopping then return false,"Belanja dihentikan." end
        syncShoppingWorld()
        if world~=app.ActiveVendingWorld and world~=app.HomeWorld then return false,"World di luar tujuan belanja / My World." end
        shopWalk.Release()
        local view=session.View()
        if view.ready and not view.picker and not view.loading and view.world==world then return true end
        if not view.ready or view.loading or view.typing then return false,"Menunggu menu world siap." end
        if os.clock()<nextShopTravel then return false,"Menunggu pindah world." end
        nextShopTravel=os.clock()+3
        if not view.picker then
            local ok,message=session.OpenPicker()
            if ok then nextShopTravel=os.clock()+.3 end
            return false,message or "Membuka pilihan world."
        end
        local ok,message=session.Enter(world)
        return false,message or (ok and "Memuat "..world.."..." or "World belum bisa dimasuki.")
    end
    shopper=makeShopping({userId=player.UserId,placeId=game.PlaceId,shopWorld=app.VendingWorld,read=readfile,write=writefile,
        encode=function(v) return http:JSONEncode(v) end,decode=function(v) return http:JSONDecode(v) end,
        spawn=task.spawn,pending=app.CommercePending,io=io})
    app.Commerce,app.Shopper,app.CommerceTravel=commerce,shopper,shopWalk
    app.CommercePending=shopper.Pending()
    syncShoppingWorld()
end
local function shoppingStep(now,view)
    if not shopper then return false end
    if not app.WantsRun then
        -- A completed native receipt can be recorded after STOP. Step(false)
        -- never starts another buy, world change or movement.
        shopper.Step(now,false)
        if type(shopper.Status)=="function" then app.ActiveVendingWorld=shopper.Status().shopWorld or app.VendingWorld end
        return false
    end
    if not app.HomeWorld then
        local plan,message=ensureHomePlan()
        if not plan then
            if shopper.Active() then status(message or "Memuat My World sebelum melanjutkan belanja.") return true end
            return false
        end
    end
    if not app.Shopping then
        local active=shopper.Active()
        if not active and (app.Recovering or not initialized or not engine or not engine.State.running) then return false end
        local ready,message=shopper.Ready(now)
        app.ShoppingStatus=message
        shopWanted=active or ready==true
        if engine and type(engine.RequestSuspend)=="function" then engine.RequestSuspend(shopWanted) end
        if not shopWanted then return false end
        if engine and engine.State.running and not engine.CanSuspend() then return false end
        if app.SeedOverflowPending and app.SeedOverflowPending.nativePending and not app.SeedOverflowPending.done then return false end
        refreshProgress() saveConfig(true)
        if engine then engine.Stop() end
        if pickup then pickup.Cancel() end
        if adapter then adapter.Release() end
        if commerce and type(commerce.RefreshWallet)=="function" then commerce.RefreshWallet() end
        app.Shopping=true app.Recovering=false
        recoveryToken=recoveryToken+1 joinPending=false recoveryBound=false arrivalAt=nil
    end
    local ok,busy,message,phase=pcall(shopper.Step,now,true)
    if type(shopper.Status)=="function" then app.ActiveVendingWorld=shopper.Status().shopWorld or app.VendingWorld end
    app.ShoppingStatus=ok and message or tostring(busy)
    if not ok then
        shopWalk.Release()
        status("Belanja tertahan: "..tostring(busy):sub(1,130))
        return true
    end
    app.ShoppingPhase=phase
    if busy then
        status(message or "Belanja item vending sampai World Lock di tas habis.")
        if now>=nextUI then nextUI=now+.2 if engine then ui.SetStats(engine.State) end end
        return true
    end
    shopWalk.Release()
    app.Shopping=false shopWanted=false initialized=false
    beginRecovery(now)
    saveConfig(true)
    return true
end
local function initialize(now)
    if initializing or now<nextInit then return end
    nextInit=now+1 initializing=true
    local ok,message=pcall(function()
        player=game.Players.LocalPlayer
        assert(player,"Menunggu akun Roblox siap.")
        -- Scenic graphics are independent of farm state and remain light on STOP.
        -- The exact Backdrop layer is restored only when this controller closes.
        if not backgroundAttempted and type(makeBackgroundOptimizer)=="function" then
            local playerGui=player:FindFirstChildOfClass("PlayerGui")
            if playerGui then
                backgroundAttempted=true
                local visualOK,result=pcall(makeBackgroundOptimizer,player,app)
                if visualOK then backgroundOptimizer=result app.BackgroundOptimizer=result end
            end
        end
        if not ui then
            if type(player.FindFirstChildOfClass)=="function" then assert(player:FindFirstChildOfClass("PlayerGui"),"Menunggu tampilan game siap.") end
            ui=makeUI(callbacks) app.UI=ui
        end
        if not session then session=makeWorldSession(player,app) app.Session=session end
        if not settingsLoaded then loadSettings() end
        initializeShopping()
        local view=session.View()
        if not app.Shopping and not initialized and not (app.WantsRun and (view.picker or not view.ready or view.loading or view.world=="")) then
            -- Bootstrap and recovery share one discovery attempt per backoff.
            if app.Recovering then nextBind=nextInit end
            initializeWorld(view)
            if initialized and app.Recovering and app.Anchor and app.LastRun and app.World==app.LastRun.world then recoveryBound=true end
        end
    end)
    initializing=false
    if not ok then status("Belum siap: "..tostring(message):sub(1,150)) end
end
local function tick()
    if not app.Alive then return end
    local now=os.clock()
    if not initialized or not settingsLoaded or not ui or not session then initialize(now) end
    if not settingsLoaded or not ui or not session then return end
    local sessionOK,view=pcall(session.View)
    if not sessionOK then view={ready=false} end
    if shoppingStep(now,view) then return end
    if recoverStep(now,view) then
        saveConfig(false)
        if now>=nextUI then nextUI=now+.2 if engine then ui.SetStats(engine.State) end end
        return
    end
    if not initialized or not engine or not pickup or not adapter then return end
    if worldChanged() then
        local rebound,why=pcall(bindCurrentWorld,true)
        if not rebound then stopRun(true) initialized=false status("Menunggu world siap: "..tostring(why):sub(1,110)) end
        return
    end
    if pickup.State.busy or pickup.State.failed then pickup.Cancel() end
    engine.Step(now)
    if app.WantsRun and not engine.State.running then
        local currentOK,current=pcall(session.View)
        if currentOK and current.picker then beginRecovery(now) return end
        app.WantsRun=false saveConfig(true)
    end
    if not adapter.SeedOverflowActive then
        local collected,failure=pcall(adapter.Collect,now)
        if not collected then adapter.LastCollect=tostring(failure):sub(1,100) end
    end
    if now>=nextUI then
        nextUI=now+.2
        local counted,seeds,blocks=pcall(adapter.Inventory)
        if counted then engine.State.seeds,engine.State.blocks=seeds,blocks end
        if not engine.State.running then
            local read,v=pcall(adapter.Snapshot)
            if read and v.ready and #v.seedSlots==adapter.SeedCount then
                local growing,mature=0,0
                for _,slot in ipairs(v.seedSlots) do
                    if slot.state=="growing" then growing=growing+1 elseif slot.state=="mature" then mature=mature+1 end
                end
                engine.State.growing,engine.State.mature,engine.State.occupied=growing,mature,growing+mature
            end
        end
        engine.State.collected=adapter.Collected
        if app.WantsRun and engine.State.running then refreshProgress() end
        ui.SetStats(engine.State) ui.SetRunning(app.WantsRun or engine.State.running)
        local displayStatus=engine.State.status
        if engine.State.running and type(adapter.LastBlockCleanup)=="string" and adapter.LastBlockCleanup:match("^Tas penuh:") then
            displayStatus=displayStatus.."\nDrop tengah tertunda: tas penuh; stok dipakai dahulu."
        end
        if engine.State.running and app.Mode=="farm" and app.Crop and app.Crop.seedId==17
            and engine.State.seeds>=180 and not adapter.SeedOverflowActive
            and type(adapter.OverflowStatus)=="string" and adapter.OverflowStatus:match("^Recycle") then
            displayStatus=displayStatus.."\n"..adapter.OverflowStatus
        end
        if shopper and type(shopper.Status)=="function" then
            local shop=shopper.Status()
            if shop.disabled or shop.phase=="blocked" then displayStatus=displayStatus.."\n"..tostring(shop.message) end
            if shop.phase=="cooldown" and shop.lastReturnReason then
                displayStatus=displayStatus.."\nBelanja: "..tostring(shop.lastReturnReason)
            end
        end
        ui.SetStatus(displayStatus)
    end
    if app.WantsRun then saveConfig(false) end
end
